def read(path):
    return json.loads(Path(path).read_text(encoding="utf-8-sig"))


def write(path, payload):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(path.suffix + ".tmp")
    temp.write_text(store.encoded(payload), encoding="utf-8")
    temp.replace(path)


def benchmark(data_root, day):
    path = data_root / "000300_SH_index.csv"
    if not path.exists():
        return {}
    f = pd.read_csv(path)
    f["day"] = pd.to_datetime(f.trade_date.astype(str)).dt.strftime("%Y-%m-%d")
    f = f[f.day.le(day)].sort_values("day").drop_duplicates("day")
    if len(f) < 21 or f.day.iloc[-1] != day:
        return {}
    c = pd.to_numeric(f.close, errors="coerce")
    return {"name": "CSI300", "as_of": day, "return5": ranking.number((c.iloc[-1] / c.iloc[-6] - 1) * 100),
            "return20": ranking.number((c.iloc[-1] / c.iloc[-21] - 1) * 100)}


def load_features(records, data_root, day):
    metrics = {}
    for i, row in enumerate(records, 1):
        try:
            metrics[row["symbol"]] = ranking.features(ranking.prices(data_root / f"{row['symbol']}_daily.csv", day), day)
        except (OSError, ValueError, KeyError, AttributeError):
            metrics[row["symbol"]] = {"status": "DATA_WAIT"}
        if i % 200 == 0:
            print(f"[trading] rank features {i}/{len(records)}", flush=True)
    return metrics


def monitor_paper(day, db=store.DB_PATH, data_root=None):
    """EOD decisions use actual paper fills; orders run at a later session's open."""
    from .. import route_evidence as RE, exit_policy as XP
    data_root = Path(data_root or store.ROOT / "data")
    with paper.database(db) as conn:
        account = conn.execute("SELECT * FROM paper_account").fetchone()
        if not account:
            return {"status": "UNCONFIGURED", "live_orders_enabled": False}
        last = conn.execute("SELECT MAX(as_of) FROM paper_valuations").fetchone()[0]
        if last and day < last:
            return {"status": "HISTORICAL_NO_ACCOUNT_REWIND", "last_day": last}
        plans = [json.loads(r[0]) for r in conn.execute("SELECT p.payload FROM paper_plans p WHERE EXISTS (SELECT 1 FROM paper_lots l WHERE l.plan_id=p.id AND l.remaining>0) OR EXISTS (SELECT 1 FROM paper_orders o WHERE o.plan_id=p.id AND o.status IN ('OPEN','PARTIAL','EXIT_BLOCKED','UNKNOWN'))")]
        frames, bars, issues = {}, {}, []
        limits = {}
        limit_path = data_root.parent / "results/intraday/reference" / f"stk_limit_{day.replace('-', '')}.json"
        try:
            ref = read(limit_path)
            if ref.get("api") == "stk_limit" and ref.get("as_of") == day and ref.get("status") in {"AVAILABLE", "PARTIAL"}:
                from ..intraday_quotes import symbol_of
                limits = {symbol_of(r["ts_code"]): ranking.number(r.get("down_limit")) for r in ref.get("records", [])}
        except (OSError, ValueError, KeyError, AttributeError, TypeError):
            pass
        index_path = data_root / "000300_SH_index.csv"
        previous_market_day, index_days = None, []
        if index_path.exists():
            index = pd.read_csv(index_path)
            index_days = sorted(set(pd.to_datetime(index.trade_date.astype(str)).dt.strftime("%Y-%m-%d")))
            earlier = [d for d in index_days if d < day]
            previous_market_day = earlier[-1] if earlier and day in index_days else None
        if plans and not previous_market_day:
            issues.append("NO_VERIFIED_MARKET_SESSION")
        for plan in plans:
            symbol = plan["symbol"]
            if symbol in frames:
                continue
            try:
                raw = ranking.prices(data_root / f"{symbol}_daily.csv", day)
                if raw.empty or raw.date.iloc[-1] != day:
                    raise ValueError("当日行情缺失")
                r = raw.iloc[-1]
                if not all((ranking.number(r.get(k)) or 0) > 0 for k in ("open", "high", "low", "close", "preclose", "volume")):
                    raise ValueError("行情价格或成交量无效")
                raw["date"] = pd.to_datetime(raw.date)
                frames[symbol] = RE.build(raw, symbol)
                since = conn.execute("SELECT MIN(entry_day) FROM paper_lots WHERE symbol=? AND remaining>0", (symbol,)).fetchone()[0]
                changed = raw.preclose.div(raw.close.shift()).sub(1).abs().gt(.001)
                if since:
                    # Inferred from price only; the ledger moves after a human confirms the event kind,
                    # and only an unconfirmed event keeps blocking the position.
                    for inferred in corporate_actions.detect(raw, since=since):
                        corporate_actions.record(conn, plan["id"], inferred)
                    corporate_action = bool(corporate_actions.pending_for(conn, plan["id"]))
                else:
                    corporate_action = bool(changed.iloc[-1])
                main_active = ranking.number(r.get("tradestatus")) == 1 and bool(paper.re.fullmatch(r"(?:sh_60|sz_00)\d{4}", symbol))
                st = ranking.number(r.get("isST"))
                if main_active and st == 1 and not (0 < (limits.get(symbol) or 0) < float(r.preclose)):
                    issues.append(f"{symbol}: ST_LIMIT_REFERENCE_WAIT")
                bars[symbol] = {"date": day, "open": float(r.open), "preclose": float(r.preclose), "raw_verified": True,
                    "previous_market_day": previous_market_day,
                    "ordinary_mainboard": main_active and st == 0 and not any(x in plan["name"].upper() for x in ("ST", "退")),
                    "risk_warning_mainboard": main_active and st == 1, "down_limit": limits.get(symbol),
                    "volume": float(r.volume),
                    "corporate_action": corporate_action}
            except (OSError, ValueError, KeyError, AttributeError) as exc:
                issues.append(f"{symbol}: {exc}")
        # A per-day checkpoint makes reruns idempotent, including partially filled exits.
        checkpoint = "PAPER_OPEN:" + day
        if not conn.execute("SELECT 1 FROM paper_events WHERE id=?", (checkpoint,)).fetchone():
            paper.process_open(conn, day, bars)
            if not issues and previous_market_day:
                paper.event(conn, "OPEN_PROCESSED", {"day": day}, checkpoint)
        market = XP.market_frame(index, day) if index_path.exists() else pd.DataFrame(
            columns=["market_strong", "market_weak", "market_ret20"], index=pd.DatetimeIndex([]))
        for plan in plans:
            pid = plan["id"]
            lots = list(conn.execute("SELECT * FROM paper_lots WHERE plan_id=? ORDER BY entry_day,id", (pid,)))
            remaining = sum(r["remaining"] for r in lots)
            if not lots or not remaining:
                continue
            f = frames.get(plan["symbol"])
            problems = []
            if f is None or f.empty:
                problems.append("PRICE_DATA_WAIT")
            # Changed exit code never re-judges an old plan; only its deterministic floors keep running.
            rules_review = plan["exit_code_hashes"] != paper.source_fingerprint()
            if rules_review:
                issues.append(f"{plan['symbol']}: FROZEN_RULE_MISMATCH_REQUIRES_REVIEW")
            if bars.get(plan["symbol"], {}).get("corporate_action"):
                problems.append("CORPORATE_ACTION_RECONCILIATION_REQUIRED")
            if not index_days or day not in index_days:
                problems.append("MARKET_CALENDAR_WAIT")
            if f is not None and not f.empty:
                available_days = set(f.date.dt.strftime("%Y-%m-%d"))
                if any(d not in available_days for d in index_days if lots[0]["entry_day"] <= d <= day):
                    problems.append("POSITION_HISTORY_GAP")
            if problems:
                valuation = {"day": day, "status": "DATA_WAIT", "remaining": remaining,
                             "reasons": problems + (["FROZEN_RULE_MISMATCH_REQUIRES_REVIEW"] if rules_review else [])}
                paper.event(conn, "POSITION_DATA_WAIT", {"plan_id": pid, **valuation})
                issues.extend(f"{plan['symbol']}: {x}" for x in problems)
            else:
                f = XP.enrich(f.reset_index(drop=True), market)
                days = f.date.dt.strftime("%Y-%m-%d").tolist()
                entry_day = lots[0]["entry_day"]
                if entry_day not in days:
                    raise ValueError(f"{pid}: 首仓行情缺失，不能重建退出依据")
                entry = days.index(entry_day)
                scale = float(f.close.iloc[entry] / f.index_close.iloc[entry])
                atr = plan["atr_cents"] / 100 / scale
                structure = plan["structure_cents"] / 100 / scale
                # Boundaries stay in the units of the entry day; after a confirmed share event every lot is in
                # current units, so lots map into the adjusted index space through the event factor.
                adjust = 1.
                for action in corporate_actions.applied(conn):
                    if action["plan_id"] == pid and action["applied"]["kind"] == "SHARE_EVENT":
                        adjust *= float(action["applied"]["factor"])
                lot_scale = scale / adjust

                def cost_on(d):
                    """Average cost of the lots filled up to day d, so history is replayed as it was."""
                    held = [r for r in lots if r["entry_day"] <= d]
                    return sum(r["quantity"] * r["price_cents"] for r in held) / sum(r["quantity"] for r in held) / 100 / lot_scale

                # The hard floor is frozen on the first fill; a cheaper add lowers the average, never the floor.
                price = cost_on(entry_day)
                hard = max(price - 2 * atr, plan["stop_cents"] / 100 / scale)
                a = {c: f[c].to_numpy() for c in f.columns}
                peak, soft, pending, breaches, required = price, hard, "", 0, 1
                sold_reasons = {json.loads(r[0])["reason"] for r in conn.execute("SELECT payload FROM paper_orders WHERE plan_id=? AND side='SELL' AND filled>0", (pid,))}
                reduced = "RISK_REDUCTION" in sold_reasons
                reduction = False
                for j in range(entry, len(f)):
                    price = cost_on(days[j])
                    if a["index_low"][j] <= hard:
                        pending = pending or "INITIAL_STOP"
                    if a["index_close"][j] < structure:
                        pending = pending or "INITIAL_STRUCTURE_FAILED"
                    if not rules_review:
                        review = XP.close_review(a, j, entry, price, atr, peak, soft_stop=soft, breaches=breaches,
                            required=required, pending_exit=pending, pending_reduce=reduction, reduced=reduced)
                        soft, breaches, required = (review[k] for k in ("soft_stop", "breaches", "required"))
                        pending, reduction = review["pending_exit"], review["pending_reduce"]
                    peak = max(peak, float(a["index_high"][j]))
                    if j - entry + 1 >= 60:
                        pending = pending or "HORIZON"
                reduction_reason = ""
                if pending:
                    paper.queue_exit(conn, pid, day, pending)
                elif not rules_review:
                    # Discretionary reductions and targets belong to the exit policy; they wait for review.
                    if reduction:
                        reduction_reason = "RISK_REDUCTION"
                        paper.queue_exit(conn, pid, day, "RISK_REDUCTION", .5)
                    elif peak >= price + 3 * atr and "TARGET_3ATR" not in sold_reasons:
                        reduction_reason = "TARGET_3ATR"
                        paper.queue_exit(conn, pid, day, "TARGET_3ATR", .5)
                    elif peak >= price + 6 * atr and "TARGET_6ATR" not in sold_reasons:
                        reduction_reason = "TARGET_6ATR"
                        paper.queue_exit(conn, pid, day, "TARGET_6ATR", .5)
                mark = paper.cents(f.close.iloc[-1])
                ratio = float(f.close.iloc[-1] / f.index_close.iloc[-1])
                valuation = {"day": day,
                             "status": ("EXIT_REQUIRED" if pending else "REDUCE_REQUIRED" if reduction_reason
                                        else "RULES_REVIEW_HOLD" if rules_review else "HOLD"),
                             "reason": pending or reduction_reason or ("FROZEN_RULE_MISMATCH_REQUIRES_REVIEW" if rules_review else ""),
                             "remaining": remaining, "mark_cents": mark, "market_value_cents": mark * remaining,
                             "hard_floor": hard * ratio, "soft_floor": soft * ratio,
                             "cost_basis_cents": int(round(price * lot_scale * 100)),
                             "deterministic_only": rules_review,
                             "policy_version": plan["exit_version"], "held_sessions": len(f) - entry}
            vid = store.digest([pid, valuation])
            conn.execute("INSERT OR IGNORE INTO paper_valuations VALUES (?,?,?,?)", (vid, day, pid, store.encoded(valuation)))
        # D07: adds are evaluated only after an actual fill, under the plan's dispatched rules.
        account, cfg = paper._account(conn)
        exposures = paper._exposure(conn)
        add_rows = []
        for plan in plans:
            dispatch = rules.dispatch(plan, db, conn=conn)
            frame = frames.get(plan["symbol"])
            if frame is None or frame.empty:
                add_rows.append({"plan_id": plan["id"], "status": "DATA_WAIT",
                                 "stop_reason": "PRICE_DATA_WAIT", "dispatch": dispatch["status"]})
                continue
            enriched = XP.enrich(frame.reset_index(drop=True), market)
            evaluation = adds.evaluate(conn, plan, enriched, index_days, day, exposures,
                                       account, cfg, dispatch["status"])
            evaluation["dispatch"] = dispatch["status"]
            if evaluation["status"] == "ADD_CONFIRMED":
                evaluation["placed"] = adds.queue(conn, plan, evaluation, account, cfg)
            add_rows.append(evaluation)
        return {"status": "DATA_WAIT" if issues else "SUCCESS", "issues": issues,
                "adds": add_rows, "live_orders_enabled": False}


def live_run(day, clock=None):
    """Books place orders only from `day`'s official close until 09:00 of the next weekday. A closure that runs
    past midnight is still that day's closure (its orders target the next open); a later rerun only reconstructs.
    A holiday on the next weekday closes the window early, which errs toward placing nothing."""
    clock = (clock or store.now())[:16]
    nxt = pd.Timestamp(day) + pd.Timedelta(days=1)
    while nxt.weekday() >= 5:
        nxt += pd.Timedelta(days=1)
    return f"{day}T15:00" <= clock < f"{nxt.date().isoformat()}T09:00"

def run(day, db=store.DB_PATH, root=store.ROOT, monitor=True):
    day = pd.Timestamp(day).date().isoformat()
    if day > store.now()[:10] or (day == store.now()[:10] and store.now()[11:16] < "15:00"):
        raise ValueError("收盘闭环不能使用未来或未收盘日期")
    lock = runner.acquire(f"after_close:{day}", db)
    if not lock["acquired"]:
        raise ValueError(f"另一次收盘运行正在进行：{lock['owner']}（{lock.get('reason', '')}）")
    # Whatever fails, the failure is on record and the lock is released; a rerun never waits out the timeout.
    try:
        return _closure(day, db, root, monitor, lock)
    except BaseException as exc:
        try:
            runner.checkpoint(day, "AFTER_CLOSE", "closure", "FAILED", {"error": str(exc)[:500]}, db)
        except Exception:
            pass
        raise
    finally:
        runner.release(f"after_close:{day}", db)


def _closure(day, db, root, monitor, lock):
    output = root / "results/mainwave_engine" / day.replace("-", "")
    data_root = root / "data"
    # D01: required inputs are checked before anything is recorded; degraded items stay named.
    schema = migrations.upgrade(db)
    startup = preflight.check(day, root, db, data_root)
    write(output / "closure_preflight.json", startup)
    runner.checkpoint(day, "AFTER_CLOSE", "preflight", "DONE", {"status": startup["status"]}, db)
    if startup["status"] == "BLOCKED":
        raise ValueError(f"启动自检未通过，缺少必要输入：{'、'.join(startup['blocking'])}")
    universe, evidence = (read(output / name) for name in ("analysis_universe.json", "analysis_evidence.json"))
    if universe["as_of"] != day:
        raise ValueError("日报日期不匹配")
    registry = knowledge.build(root)
    registry_id = knowledge.persist(registry, db)
    rule_set = rules.register(db, root)
    with store.connect(db) as conn:
        prior = store.latest(conn, (pd.Timestamp(day) - pd.Timedelta(days=1)).date().isoformat())
    leaders = {}
    path = output / "leader_state_daily.csv"
    if path.exists():
        f = pd.read_csv(path).fillna("")
        for row in f.to_dict("records"):
            if row.get("as_of") == day:
                leaders.setdefault(row.get("leader"), []).append(row)
    payload = ranking.build(universe, evidence, load_features(universe["records"], data_root, day), benchmark(data_root, day), prior, leaders)
    if day == store.now()[:10]:
        payload["observation_basis"] = "FORWARD_RECORDED_AFTER_CLOSE"
    payload["knowledge_registry_id"] = registry_id
    payload["code_sha256"] = hashlib.sha256(Path(ranking.__file__).read_bytes()).hexdigest()
    payload["source_hashes"] = {name: hashlib.sha256((output / name).read_bytes()).hexdigest() for name in ("analysis_universe.json", "analysis_evidence.json")}
    snap = store.persist(payload, db)
    write(root / "results/trading_system/knowledge_registry.json", registry)
    write(output / "trading_closure.json", snap)
    sessions = preflight._calendar(data_root, day)[0] or []
    # D02-D05: continuity, per-step route cards, rank explanation and forward validation.
    chain = continuity.advance(snap, db, sessions)
    write(output / "opportunity_continuity.json", chain)
    # Shadow layer: layered screening, market policy dimensions, entry-contract drafts, discovery register.
    # None of it changes states, ranks or permissions; a failure here never blocks the closure.
    shadow, watch = None, []
    try:
        prior_day = str((prior or {}).get("as_of", "")).replace("-", "")
        prior_shadow_path = root / "results/mainwave_engine" / prior_day / "ranking_shadow.json"
        watch_path = output / "preparation_watch.csv"
        watch = pd.read_csv(watch_path, encoding="utf-8-sig").to_dict("records") if watch_path.exists() else []
        prior_facts = shadow_inputs.previous_review_facts(root, prior_day)
        shadow = ranking_shadow.evaluate(snap, read(prior_shadow_path) if prior and prior_shadow_path.exists() else None, watch,
                                         theme_facts=prior_facts.get("themes"))
        write(output / "ranking_shadow.json", shadow)
        policy = market_policy.evaluate(snap.get("market", {}), (prior_facts.get("sentiment") or {}).get("metrics"),
                                        (prior_facts.get("loss_effects") or {}).get("events"), prior_facts.get("as_of"))
        write(output / "market_policy.json", policy)
        drafts = shadow_inputs.contract_drafts(snap, day, sessions, output)
        write(output / "entry_contract_drafts.json", drafts)
        shadow_summary = {"status": "SUCCESS", **{k: shadow["summary"][k] for k in ("total", "re_eligible", "cohort_strong", "top10_overlap_with_frozen")},
                          "review_shortlist": [x["symbol"] for x in shadow["summary"]["review_shortlist"]],
                          "preparation_watch": shadow["summary"]["preparation_watch"], "contract_drafts": drafts["summary"],
                          "market_policy": {e: policy["policy"].get(e, {}).get("shadow") for e in ("E1", "E2", "E3")}}
    except Exception as exc:  # a shadow must never block the closure
        shadow_summary = {"status": "FAILED", "error": str(exc)[:300]}
    cards = diagnostics.report(snap)
    write(output / "route_diagnostics.json", cards)
    board = explain.board(snap, prior)
    write(output / "ranking_explained.json", board)
    runner.renew(f"after_close:{day}", db)
    review = validation.evaluate(day, db, data_root)
    write(output / "ranking_validation.json", review)
    audit = {"protocols": validation.protocols(root), "ablation": validation.ablate(day, root),
             "missed": validation.missed_audit(day, db, data_root)}
    write(output / "validation_audit.json", audit)
    runner.renew(f"after_close:{day}", db)
    paper_status = monitor_paper(day, db, data_root) if monitor else {"status": "NOT_RUN"}
    # Three books apart: SIGNAL records the mechanical trade set, AUTO executes it without a person,
    # MANUAL is this ledger. Only a same-day run lets the auto book place orders.
    signal = books.record_signal(day, snap, db)
    auto = (books.run_auto(day, payload, snap, root, data_root, live=live_run(day), signal=signal, main_db=db)
            if monitor else {"status": "NOT_RUN", "book": "AUTO"})
    try:
        new100k = (books.run_new100k(day, payload, snap, root, data_root, live=live_run(day), signal=signal, main_db=db)
                   if monitor else {"status": "NOT_RUN", "book": "NEW100K"})
    except Exception as exc:  # the compounding book must never block the closure or the other books
        new100k = {"status": "FAILED", "book": "NEW100K", "error": str(exc)[:300]}
    try:
        register = shadow_inputs.discovery_register(root, day, snap, shadow, watch, auto, sessions)
        write(output / "discovery_events.json", register)
        shadow_summary["discovery"] = register["summary"]
    except Exception as exc:  # the register is an observation artefact; its failure is recorded, not raised
        shadow_summary["discovery"] = {"status": "FAILED", "error": str(exc)[:300]}
    status = {"as_of": day, "version": VERSION, "snapshot_id": snap["snapshot_id"], "ranking": snap["summary"],
              "validation": review["summary"], "knowledge": registry["summary"], "paper": paper_status,
              "preflight": {"status": startup["status"], "blocking": startup["blocking"],
                            "degraded": startup["degraded"]},
              "continuity": chain["summary"], "diagnostics": cards["summary"], "shadow_screening": shadow_summary,
              "explained": board["summary"],
              "audit": {"ablation": audit["ablation"]["status"], "missed": audit["missed"]["status"]},
              "rules": {"fingerprint": rule_set["fingerprint"], "software": rule_set["software_version"]},
              "live_orders_enabled": False, "status": "PARTIAL" if paper_status["status"] == "DATA_WAIT" else "SUCCESS"}
    perf = performance.report(day, db)
    write(output / "paper_performance.json", perf)
    manual = {"book": "MANUAL", "status": paper_status["status"], "performance": perf["status"],
              **({k: perf["account"][k] for k in ("equity_cents", "net_return", "max_drawdown", "marked")}
                 if perf.get("account") else {})}
    write(output / "books_daily.json", {"version": books.VERSION, "as_of": day, "books": books.BOOKS,
                                        "signal": signal, "auto": auto, "manual": manual, "new100k": new100k,
                                        "separation": "三本账各自回答一个问题，成绩不合并。"})

def _sha(path):

def import_package(source, root=store.ROOT):
    """Install a delivered package only when every hash matches the manifest."""
    source, root = Path(source), Path(root)
    book = json.loads((source / "manifest.json").read_text(encoding="utf-8-sig"))
    register = source / Path(book["register"]).name
    if _sha(register) != book["register_sha256"]:
        raise ValueError("登记册哈希不匹配，拒绝导入")
    for item in book["sources"]:
        if _sha(source / item["path"]) != item["sha256"]:
            raise ValueError(f"来源哈希不匹配，拒绝导入：{item['path']}")
    shutil.copyfile(register, root / book["register"])
    for item in book["sources"]:
        target = root / item["path"]
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(source / item["path"], target)
    return {"version": VERSION, "imported": len(book["sources"]), "register": book["register"]}


def _calendar(data_root, day):
    path = data_root / "000300_SH_index.csv"
    if not path.exists():
        return None, _check("calendar", "交易日历与基准指数", REQUIRED, False,
                            "缺少 000300_SH_index.csv", "无法判定交易日、缺日与基准超额")
    try:
        dates = pd.read_csv(path, usecols=["trade_date"], dtype=str).trade_date
    except (OSError, ValueError) as exc:
        return None, _check("calendar", "交易日历与基准指数", REQUIRED, False,

    days = sorted(set(parsed.dropna().dt.strftime("%Y-%m-%d")))
    if day not in days:
        return days, _check("calendar", "交易日历与基准指数", REQUIRED, False,
                            f"{path.name} 缺少 {day}，本地最新有效日期：{days[-1] if days else '无'}",

def _contract_check(data_root, symbol, day):
    path = data_root / f"{symbol}_daily.csv"
    if not path.exists():
        return _check("contract", "行情单位与复权基准", REQUIRED, False, f"缺少样本行情 {symbol}",
                      "无法核对成交量、成交额、换手与复权基准")
    frame = pd.read_csv(path).tail(20)
    flags = set(pd.to_numeric(frame.adjustflag, errors="coerce").dropna().astype(int))
    row = frame.iloc[-1]
    amount, volume, close = float(row.amount), float(row.volume), float(row.close)
    consistent = volume > 0 and abs(amount / (volume * close) - 1) < .2
    turnover = pd.to_numeric(frame.turnover, errors="coerce").dropna()
    percent = bool(len(turnover)) and float(turnover.max()) <= 100
    ok = flags == {CONTRACT["adjust_flag"]} and consistent and percent
    return _check("contract", "行情单位与复权基准", REQUIRED, ok,
                  f"adjustflag={sorted(flags)}；amount≈volume×close：{consistent}；换手为百分比：{percent}",
                  "" if ok else "单位或复权基准与契约不符，价格与成交口径需人工核对")


def _bars(data_root, symbols, day, days):
    """Stale, missing and mid-series gaps are reported apart; none of them is a zero."""
    missing, stale, gapped, ready = [], [], [], 0
    window = [d for d in (days or []) if d <= day][-20:]
    for symbol in symbols:
        path = data_root / f"{symbol}_daily.csv"
        try:
            dates = set(pd.read_csv(path, usecols=["date"]).date.astype(str).str[:10])
        except (OSError, ValueError, KeyError):
            missing.append(symbol)
            continue
        if day not in dates:
            stale.append(symbol)
        elif window and [d for d in window if d not in dates]:
            gapped.append(symbol)
        else:
            ready += 1
    return missing, stale, gapped, ready

def trade_set(payload, rule=AUTO_RULE):
    """The mechanical selection from the snapshot alone; an empty set is a valid answer, not a gap.

    The rank bound applies to the original board: an excluded first place leaves two names, never three.

            and r[rule["rank"]] <= rule["top_n"] and _scope(r) not in rule["exclude_alpha_scope"]]
    rows.sort(key=lambda r: (r[rule["rank"]], r["symbol"]))
    return [{"symbol": r["symbol"], "name": r.get("name"), "opportunity_id": r["opportunity_id"],
             "participation_rank": r.get("participation_rank"), "strength_rank": r.get("strength_rank"),

def _ledger_evidence(conn, expected=None):
    """What the ledger actually did, read back from it: effective config, orders vs fills, add refusals."""
    expected = {k: v for k, v in (expected or AUTO_ACCOUNT).items() if k in ("capital", "min_fee", *ACCOUNT_KEYS)}
    from collections import Counter
    row = conn.execute("SELECT initial_cents, config FROM paper_account").fetchone()
    cfg = json.loads(row["config"])
    effective = {"capital": row["initial_cents"] / 100, "min_fee": cfg["min_fee_cents"] / 100,
                 **{k: cfg[k] for k in ACCOUNT_KEYS}}
    matches = all(float(effective[k]) == float(expected[k]) for k in expected)
    orders = [{"symbol": json.loads(o["plan"])["symbol"], "order_id": o["id"], "ordered": o["quantity"],
               "filled": o["filled"], "status": o["status"]}
              for o in conn.execute("SELECT o.*, p.payload AS plan FROM paper_orders o JOIN paper_plans p ON p.id=o.plan_id "
                                    "WHERE o.side='BUY' ORDER BY o.created_at")]
    cancelled = {json.loads(r[0])["order_id"]: json.loads(r[0])["reason"]
                 for r in conn.execute("SELECT payload FROM paper_events WHERE kind='ORDER_CANCELLED'")}
    for o in orders:
        o["unfilled_reason"] = cancelled.get(o["order_id"], "")
    add_refusals = Counter(json.loads(r[0]).get("reason", "") for r in conn.execute(
        "SELECT payload FROM paper_events WHERE kind='ADD_REJECTED'"))
    counts = {"plans": conn.execute("SELECT COUNT(*) FROM paper_plans").fetchone()[0],
              "fills": conn.execute("SELECT COUNT(*) FROM paper_fills").fetchone()[0],
              "open_positions": conn.execute("SELECT COUNT(DISTINCT plan_id) FROM paper_lots WHERE remaining>0").fetchone()[0]}
    return {"effective_account": effective, "account_matches_code": matches,
            "config_fingerprint": store.digest([effective, AUTO_RULE]),
            "executions": orders, "add_refusals": dict(add_refusals), **counts}


def _ensure_snapshot(auto_db, main_db, snapshot_id):
    """The frozen signal's snapshot must be on the auto ledger; a crashed first run may have left it out."""
    with store.connect(auto_db) as conn:
        if conn.execute("SELECT 1 FROM snapshots WHERE id=?", (snapshot_id,)).fetchone():
            return
    with store.connect(main_db) as conn:
        stored = store.snapshot(conn, snapshot_id)
    original = {k: v for k, v in stored.items() if k not in ("snapshot_id", "recorded_at")}
    store.persist(original, auto_db)


def run_auto(day, payload, snap, root=store.ROOT, data_root=None, at=None, live=True, signal=None, main_db=None):
    """AUTO book: existing positions first (fills, exits, adds), then the frozen trade set; no human input.

    Only a same-day after-close run touches the ledger. A reconstruction reads the ledger and records what
    the rule would have chosen; it matches nothing, values nothing and places nothing.
    """
    from . import paper, daily, migrations, performance, runner
    root = Path(root)
    db = root / AUTO_DB
    migrations.upgrade(db)
    with paper.database(db) as conn:
        configured = conn.execute("SELECT 1 FROM paper_account").fetchone() is not None
    if not configured:
        paper.initialize(AUTO_ACCOUNT, db, at=at)
    # The day's first publication decides; a same-day rerun executes the frozen set, not its own.
    frozen = signal or {"snapshot_id": snap["snapshot_id"], "trade_set": trade_set(snap)}
    chosen = frozen["trade_set"]
    base = {"version": VERSION, "book": "AUTO", "day": day, "db": str(db), "rule": AUTO_RULE, "live": live,

    if not live:
        with paper.database(db) as conn:

        return {**base, "ledger_touched": False,

    if frozen["snapshot_id"] == snap["snapshot_id"]:

        store.persist(payload, db)

def current_drawdown(perf):
    """Drawdown of the last marked equity from its running peak, in percent; None while nothing is marked."""
    marked = [p["equity_cents"] for p in (perf or {}).get("nav") or [] if p.get("equity_cents")]
    if not marked:
        return None
    return round((1 - marked[-1] / max(marked)) * 100, 3)

def rebound_basket(day, data_root):
    """Crash rebound basket (P2). Report only: it never creates a plan, an order or a quantity."""
    base = {"index": "中证1000", "rule": REBOUND_RULE, "report_only": True}
    try:
        f = _index_frame(day, data_root)
    except (OSError, ValueError, KeyError) as exc:

    if len(f) < 60:
        return {**base, "level": "UNKNOWN", "text": "中证1000 历史不足 60 个交易日，反弹篮子状态未知"}
    c = f["close"].astype(float)
    dd = (c / c.rolling(60).max() - 1).to_numpy()
    dates, n = f["date"].tolist(), len(f)
    facts = {"as_of": dates[-1], "close": round(float(c.iloc[-1]), 2), "drawdown_60d": round(float(dd[-1]), 4)}

    start = starts[-1] if starts and n - 1 - starts[-1] < REBOUND_WINDOW else None
    if start is None:
        return {**base, **facts, "level": "INACTIVE",
                "text": f"大跌反弹篮子未触发：中证1000 距 60 日最高收盘 {dd[-1]:.1%}（第一档在 −15%）"}
    tranches = []
    for i, lvl in enumerate(REBOUND_LEVELS, 1):
        k = next((k for k in range(start, n) if dd[k] <= lvl), None)
        hold_end = None if k is None else dates[k + 21] if k + 21 < n else "买入后第 20 个交易日"
        tranches.append({"tranche": i, "level": lvl, "signal_day": dates[k] if k is not None else None, "hold_until": hold_end})
    today = [t for t in tranches if t["signal_day"] == day]
    done = "；".join(f"第{t['tranche']}档({t['level']:.0%}) " + (f"{t['signal_day']} 触发，持有至 {t['hold_until']}" if t["signal_day"] else "未到")
                    for t in tranches)
    window = f"本轮起点 {dates[start]}，再过 {REBOUND_WINDOW - (n - 1 - start)} 个交易日未到的档位作废"
    out = {**base, **facts, "leg_start": dates[start], "tranches": tranches}
    if today:
        t = today[-1]
        return {**out, "level": "TRIGGERED_TODAY", "basket": beaten_basket(day, data_root),
                "text": f"大跌反弹篮子第 {t['tranche']} 档今日触发：中证1000 距 60 日高 {dd[-1]:.1%}。"
                        f"明日收盘买入 1/3，持有 20 个交易日（仅提示，不下单）。{done}。{window}。"}
    return {**out, "level": "ACTIVE", "text": f"大跌反弹篮子进行中：距 60 日高 {dd[-1]:.1%}。{done}。{window}。"}


def _valuations(conn, pid, day):
    rows = [(r["as_of"], json.loads(r["payload"])) for r in conn.execute(
        "SELECT as_of, payload FROM paper_valuations WHERE plan_id=? AND as_of<=? ORDER BY as_of, rowid", (pid, day))]
    today = next((v for d, v in reversed(rows) if d == day), None)
    before = next((v for d, v in reversed(rows) if d < day), None)
    return today, before


def assess(conn, plan, day, cfg, data_root):
    from . import corporate_actions
    pid = plan["id"]
    lots = [dict(r) for r in conn.execute("SELECT * FROM paper_lots WHERE plan_id=? ORDER BY entry_day, id", (pid,))]
    held = sum(l["remaining"] for l in lots)
    if not held:
        return None
    today, before = _valuations(conn, pid, day)
    bar = bar_state(plan["symbol"], day, data_root)
    if corporate_actions.pending_for(conn, pid):
        bar = {**bar, "data_state": "CORPORATE_ACTION_PENDING", "reason": "公司行动待确认；股数/价格/保护线未对齐前不做风险授权"}
    # Protection already in force today comes from an earlier close; today's close only sets the next one.
    effective = profit_tracker.protection_of(plan, before or {})
    upcoming = profit_tracker.protection_of(plan, today) if today and today.get("status") != "DATA_WAIT" else None
    hard = plan.get("stop_cents")
    exits = [dict(r) for r in conn.execute(
        "SELECT * FROM paper_orders WHERE plan_id=? AND side='SELL' AND status IN ('OPEN','PARTIAL','EXIT_BLOCKED','UNKNOWN')", (pid,))]
    pending_exit = sum(o["quantity"] - o["filled"] for o in exits)
    sellable = sum(l["remaining"] for l in lots if l["entry_day"] < day)
    statuses = {o["status"] for o in exits}
    trade_state = ("EXIT_BLOCKED" if statuses & {"EXIT_BLOCKED", "UNKNOWN"} else "EXIT_PARTIAL" if "PARTIAL" in statuses
                   else "EXIT_PENDING" if exits else "OPEN")
    # Trend: the closure's frozen exit policy is authoritative; the monitor only adds the protection checks.
    trend, trend_basis = "UNKNOWN", "今日收盘闭环尚未对此持仓估值"
    if today:
        trend = TREND_FROM_VALUATION.get(today.get("status"), "UNKNOWN")
        trend_basis = f"收盘闭环（{today.get('policy_version') or plan.get('exit_version')}）：{today.get('status')} {today.get('reason') or ''}".strip()
    breach = None
    if bar["data_state"] == "FRESH":
        if hard and round(bar["low"] * 100) <= hard:
            breach = f"日内最低 {bar['low']:.2f} 触及初始硬止损 {hard / 100:.2f}"
        elif effective and round(bar["close"] * 100) < effective:
            breach = f"收盘 {bar['close']:.2f} 跌破此前已生效保护 {effective / 100:.2f}"
        if breach:
            trend, trend_basis = "BROKEN", breach + ("（收盘闭环未处理，待闭环按冻结合同登记退出）" if not today else "")
        elif not today:
            trend, trend_basis = "INTACT", "仅核对了已生效保护线与硬止损；冻结退出合同的完整判断待收盘闭环"
    mark = None
    if bar["data_state"] == "FRESH":
        mark = (round(bar["close"] * 100), day, "MONITOR_DAILY_CLOSE")
    c = profit_tracker.campaign(conn, plan, day, cfg, upcoming or effective, mark_override=mark)
    ratio = c.get("giveback_ratio")
    pnl = c.get("net_pnl_cents")
    profit_state = ("UNKNOWN" if pnl is None else
                    "GIVEBACK_FROM_PEAK" if c.get("giveback_alert") else
                    "NEW_PEAK" if c["observed_peak_cents"] and pnl >= c["observed_peak_cents"] else
                    "NEAR_BREAKEVEN" if abs(pnl) <= (c.get("estimated_exit_cost_cents") or 0) else
                    "NET_PROFIT" if pnl > 0 else "NET_LOSS")
    # Action priority (design §04.4): hard risk / pending exit > trend exit > reduce > data > review > hold.
    blocks = []
    if trade_state != "OPEN":
        action = "EXIT_REQUIRED"
        reason = "已有退出委托未完成" + ("，当前受阻，敞口仍计入风险" if trade_state == "EXIT_BLOCKED" else "")
    elif trend == "BROKEN":
        action, reason = "EXIT_REQUIRED", trend_basis
    elif today and today.get("status") == "REDUCE_REQUIRED":
        action, reason = "REDUCE_REQUIRED", f"原退出合同要求减仓：{today.get('reason')}"
    elif bar["data_state"] != "FRESH":
        action, reason = "MANUAL_REVIEW", f"数据状态 {bar['data_state']}：{bar.get('reason', '')}；保留最后可信值，停止依赖它的新风险授权"
    elif trend in ("DEGRADED", "UNKNOWN") or c.get("giveback_alert"):
        action = "HOLD_REVIEW"
        reason = "；".join(x for x in (trend_basis if trend != "INTACT" else "",
                                       f"利润回吐 {ratio:.0%}（仅告警，不自动卖出）" if c.get("giveback_alert") else "") if x)
    else:
        action, reason = "HOLD", "原交易依据未被价格证伪，按原合同持有"
    if action in ("EXIT_REQUIRED", "REDUCE_REQUIRED"):
        blocks.append("存在退出/减仓要求")
    if bar["data_state"] != "FRESH":
        blocks.append("数据未就绪")
    if trend != "INTACT":
        blocks.append(f"趋势状态 {trend}")
    if bar.get("suspended"):
        blocks.append("停牌")
    next_event = ("下一可交易开盘按退出委托撮合；跌停/停牌则记为退出受阻并继续跟踪" if action == "EXIT_REQUIRED" else
                  f"收盘跌破 {(upcoming or effective or 0) / 100:.2f} 即触发退出" if (upcoming or effective) else "等待保护线建立")
    return {"campaign_id": pid, "symbol": plan["symbol"], "name": plan.get("name"), "theme": plan.get("theme"),
            "trade_state": trade_state, "trend_state": trend, "data_state": bar["data_state"],
            "execution_state": {"held": held, "sellable": sellable, "t_plus_1_locked": held - sellable,
                                "pending_exit_qty": pending_exit, "suspended": bool(bar.get("suspended")),
                                "exit_orders": [{"id": o["id"], "status": o["status"], "left": o["quantity"] - o["filled"]} for o in exits]},
            "profit_state": profit_state, "action": action, "action_word": ACTION_WORDS[action], "action_reason": reason,
            "add_blocked_by": blocks,
            "six_questions": {
                "why_bought": {"entry_day": c["entry_day"], "mode": plan.get("mode"), "setup": plan.get("setup_kind"),
                               "routes": plan.get("route_ids"), "still_valid": trend_basis},
                "holding": {"held": held, "sellable": sellable, "locked_t1": held - sellable},
                "protection": {"initial_hard_cents": hard, "effective_today_cents": effective,
                               "next_session_cents": upcoming,
                               "next_effective": "下一交易日" if upcoming else None},
                "risk": {k: c.get(k) for k in ("net_pnl_if_exit_at_protection_cents", "net_pnl_if_stress_exit_cents",
                                               "current_wealth_loss_in_stress_cents", "principal_loss_in_stress_cents",
                                               "stress_basis")},
                "add": "禁止：" + "、".join(blocks) if blocks else "未被禁止；是否加仓只由该计划冻结的加仓合同在收盘闭环评估",
                "action": {"action": action, "next_event": next_event}},
            "profit": c, "bar": bar}


def run(day, data_root=None, ledgers=None):
    day = pd.Timestamp(day).date().isoformat()
    data_root = Path(data_root or store.ROOT / "data")
    result = {"version": VERSION, "as_of": day, "computed_at": store.now(), "live_orders_enabled": False,
              "notice": "独立持仓监控：只读账本与本地日线，不依赖当日选股流水线；不写数量与现金。" + profit_tracker.NOTICE,
              "market_risk": market_risk(day, data_root), "market_warning": market_warning(day, data_root),
              "rebound_basket": rebound_basket(day, data_root), "etf_dip": etf_dip.alert(day, data_root),
              "books": {}, "alerts": []}
    for book, db in (ledgers or books()).items():
        if not Path(db).exists():
            result["books"][book] = {"status": "NO_LEDGER"}
            continue
        try:
            with paper.database(db) as conn:
                raw = conn.execute("SELECT * FROM paper_account").fetchone()
                if not raw:
                    result["books"][book] = {"status": "UNCONFIGURED"}
                    continue
                cfg = json.loads(raw["config"])
                plans = [json.loads(r[0]) for r in conn.execute("SELECT payload FROM paper_plans ORDER BY created_at")]
                positions = [p for p in (assess(conn, plan, day, cfg, data_root) for plan in plans) if p]
                pending_buys = [{"order_id": o["id"], "symbol": json.loads(o["plan"])["symbol"], "status": o["status"],
                                 "quantity": o["quantity"] - o["filled"], "reserve_cents": o["reserve_cents"]}
                                for o in conn.execute("SELECT o.*, p.payload AS plan FROM paper_orders o JOIN paper_plans p ON p.id=o.plan_id "
                                                      "WHERE o.side='BUY' AND o.status IN ('OPEN','PARTIAL','UNKNOWN')")]
            ledger = paper.view(db)
            profits = profit_tracker.report(db, day)
            result["books"][book] = {"status": "READY", "reconciled": ledger["reconciled"], "positions": positions,
                                     "pending_buys": pending_buys, "account": profits["account"],
                                     "closed_campaigns": [c for c in profits["campaigns"] if c["trade_closed"]]}
            if not ledger["reconciled"]:
                result["alerts"].append({"level": "P0", "book": book, "text": "账实不符：现金或股数与成交不守恒，停止新增风险"})
            for p in positions:
                level = ("P0" if p["trade_state"] == "EXIT_BLOCKED" else "P1" if p["action"] in ("EXIT_REQUIRED", "REDUCE_REQUIRED", "MANUAL_REVIEW")
                         else "P2" if p["action"] == "HOLD_REVIEW" else None)
                if level:
                    result["alerts"].append({"level": level, "book": book, "symbol": p["symbol"], "name": p["name"],
                                             "action": p["action_word"], "text": p["action_reason"]})
        except Exception as exc:  # one broken ledger never hides the others
            result["books"][book] = {"status": "FAILED", "error": str(exc)[:300]}
            result["alerts"].append({"level": "P0", "book": book, "text": f"账本读取失败：{str(exc)[:120]}"})
    result["alerts"].sort(key=lambda a: a["level"])
    result["status"] = "ALERT" if any(a["level"] == "P0" for a in result["alerts"]) else "READY"
    return result

def _frame(as_of: str, data_root) -> pd.DataFrame:
    path = Path(data_root) / INDEX_FILE
    f = pd.read_csv(path)
    f["date"] = pd.to_datetime(f["trade_date"].astype(str)).dt.strftime("%Y-%m-%d")
    f["close"] = pd.to_numeric(f["close"], errors="coerce")
    f = f.dropna(subset=["date", "close"]).sort_values("date")
    return f[f["date"] <= as_of].drop_duplicates("date").reset_index(drop=True)


def regime(as_of: str, data_root=None, span: int = SPAN,
           lookback: int = LOOKBACK) -> dict:
    """核心指数中期趋势环境。返回的 regime 取 UP / DOWN / UNKNOWN。

    2026-11-24 起只报告不拦截: blocks_entries 恒为 False, gate 恒为 ALLOW。
    """
    base = {"as_of": as_of, "index": INDEX_NAME, "version": VERSION, "rule": RULE,
            "evidence": EVIDENCE, "report_only": REPORT_ONLY,
            "blocks_entries": BLOCKS_ENTRIES,
            "gate": "ALLOW", "regime": "UNKNOWN", "status": "UNKNOWN",
            "data_date": "", "close": None, "ema": None, "slope": None, "text": ""}
    try:
        f = _frame(as_of, data_root or Path("data"))
    except (OSError, ValueError, KeyError, pd.errors.EmptyDataError) as exc:
        return {**base, "status": "MISSING",
                "text": f"{INDEX_NAME} 日线不可用 ({str(exc)[:80]}), 中期趋势未知"}
    if len(f) < MIN_BARS:
        return {**base, "status": "INSUFFICIENT_HISTORY", "data_date": str(f["date"].iloc[-1]) if len(f) else "",
                "text": f"{INDEX_NAME} 历史不足 {MIN_BARS} 个交易日, 中期趋势未知"}
    c = f["close"].astype(float).reset_index(drop=True)
    if not c.gt(0).all() or not c.map(math.isfinite).all():
        return {**base, "status": "INVALID_PRICE", "data_date": str(f["date"].iloc[-1]),
                "text": f"{INDEX_NAME} 日线收盘价无效，中期趋势未知"}
    ema = c.ewm(span=span, adjust=False).mean()
    slope = float(ema.iloc[-1] / ema.iloc[-1 - lookback] - 1)
    close = float(c.iloc[-1])
    data_date = str(f["date"].iloc[-1])
    fresh = data_date == as_of
    if not fresh:
        return {**base, "status": "STALE", "data_date": data_date, "close": round(close, 2),
                "ema": round(float(ema.iloc[-1]), 2), "slope": round(slope, 4),
                "text": f"{INDEX_NAME} 最新日线为 {data_date}, 不是 {as_of}, 中期趋势未确认"}
    up = slope > 0
    return {**base, "status": "FRESH", "data_date": data_date, "close": round(close, 2),
            "ema": round(float(ema.iloc[-1]), 2), "slope": round(slope, 4),
            "regime": "UP" if up else "DOWN",
            "text": (f"{INDEX_NAME} EMA{span} 中期趋势向上 (10 日斜率 {slope*100:+.2f}%)"
                     if up else
                     f"{INDEX_NAME} EMA{span} 中期趋势向下 (10 日斜率 {slope*100:+.2f}%)")
                    + "（只报告, 不拦截 S4 候选）"}


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--date", required=True)
    ap.add_argument("--data-root", default="data")
    args = ap.parse_args()
    out = regime(args.date, args.data_root)
    print(json.dumps(out, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()

def checked_date(value):
    day = date.fromisoformat(value)
    if day.isoformat() != value or day > datetime.now(timezone(timedelta(hours=8))).date():
        raise ValueError("需要已经结束的行情日期，格式 YYYY-MM-DD")
    now = datetime.now(timezone(timedelta(hours=8)))
    if day == now.date() and now.hour < 15:
        raise ValueError("今日尚未收盘，请使用已收盘交易日")
    return value

def add(symbol, as_of, db=DB_PATH, strategy=None, meta=None):
    if strategy is not None and strategy not in STRATEGIES:
        raise ValueError(f"未知策略：{strategy}")
    symbol = SA.normalize_symbol(symbol)
    # Fast retry path also preserves the original price/date after later daily updates.
    with closing(connect(db)) as conn:
        old = conn.execute("SELECT payload FROM positions WHERE symbol=? AND status='OPEN'", (symbol,)).fetchone()
        if old:
            return {"position": json.loads(old[0]), "created": False}
    _, quote = close_quote(symbol, as_of)
    board = SA._read_json(SA.OUT_ROOT / as_of.replace("-", "") / "analysis_universe.json", {})
    context = next((r for r in board.get("records", []) if r["symbol"] == symbol), {})
    identity = SA._mapping(symbol)
    row = {"id": uuid4().hex, "symbol": symbol, "name": context.get("name") or identity.get("name") or symbol,
           "theme": context.get("belonging_theme", ""), "route_ids": context.get("route_ids", ""),
           "entry_date": as_of, "entry_price": float(quote.close),
           "atr_fraction": float(quote.atr20 / quote.index_close), "status": "OPEN",
           "source": "MANUAL_CLOSE", "created_at": datetime.now(timezone.utc).isoformat(),
           "version": VERSION, "exit_policy_version": LAD_VERSION if strategy == "S4_LAD" else XP.VERSION}
    if strategy:
        row["strategy"] = strategy
    row.update({k: v for k, v in (meta or {}).items() if k.startswith("s4_")})
    if 2 * row["atr_fraction"] >= 1:
        raise ValueError("初始风险底线无效，不能建立虚拟持仓")
    with closing(connect(db)) as conn, conn:
        conn.execute("INSERT OR IGNORE INTO positions VALUES (?, ?, ?, ?, ?)",

        stored = json.loads(conn.execute("SELECT payload FROM positions WHERE symbol=? AND status='OPEN'", (symbol,)).fetchone()[0])
    return {"position": stored, "created": stored["id"] == row["id"]}

def evaluate(position, frame, as_of, calendar, market, g1=frozenset()):
    """Rebuild advice from the actual paper close entry, never a signal's hypothetical open."""
    base = {"as_of": as_of, "version": VERSION, "policy_version": XP.VERSION,
            "status": "DATA_WAIT", "reason": "行情缺失或不连续，不能判为持有安全", "events": []}
    f = frame[frame.date.le(as_of)].copy().reset_index(drop=True)
    if f.empty:
        return base
    days = f.date.dt.strftime("%Y-%m-%d").tolist()
    base.update(price_date=days[-1], close=float(f.close.iloc[-1]))
    entry_day = position["entry_date"]
    if entry_day not in days:
        return base
    entry = days.index(entry_day)
    if not math.isclose(float(f.close.iloc[entry]), position["entry_price"], rel_tol=1e-6):
        return {**base, "reason": "入场日原始行情已修订，请核对原始成本；不自动覆盖台账"}
    present = set(days) | set(frame.attrs.get("suspended_days", []))
    needed = [d for d in calendar if entry_day <= d <= as_of]
    if not needed or needed[-1] != as_of or any(d not in present for d in needed):
        return base
    relevant = f.iloc[entry:][["index_open", "index_high", "index_low", "index_close", "close"]]
    if relevant.isna().any().any() or not relevant.map(lambda x: math.isfinite(x) and x > 0).all().all():
        return base
    if position.get("strategy") == "S4_LAD":
        return lad_review(f, entry, days, as_of, base, position, g1)
    f = XP.enrich(f, market)
    a = {c: f[c].to_numpy() for c in f.columns}
    price = float(f.index_close.iloc[entry])
    atr = price * position["atr_fraction"]
    hard = soft = price - 2 * atr
    peak, pending, reduction, breaches, required = price, "", False, 0, 1
    events, seen = [], set()
    initial_warning, _ = XP.warning(a, entry, entry, price, atr, price)
    if initial_warning == "EXIT_STRUCTURE":
        pending = initial_warning
        events.append({"date": entry_day, "reason": pending, "label": XP.REASONS[pending]})
        seen.add(pending)
    # Close entry: the entry day's earlier low/high cannot trigger a stop or profit target.
    for j in range(entry + 1, len(f)):
        reasons = []
        if a["index_low"][j] <= hard:
            pending = "INITIAL_STOP"
        review = XP.close_review(a, j, entry, price, atr, peak, soft_stop=soft, breaches=breaches,
                                 required=required, pending_exit=pending, pending_reduce=reduction)
        soft, breaches, required = (review[k] for k in ("soft_stop", "breaches", "required"))
        pending, reduction = review["pending_exit"], review["pending_reduce"]
        if j - entry + 1 >= 60:
            pending = pending or "HORIZON"
        if pending:
            reasons.append(pending)
        elif reduction:
            reasons.append("RISK_REDUCTION")
        for multiple in (3, 6):
            if a["index_high"][j] >= price + multiple * atr:
                reasons.append(f"TARGET_{multiple}ATR")
                reduction = True
        for reason in reasons:
            if reason not in seen:
                events.append({"date": days[j], "reason": reason,
                               "label": XP.REASONS.get(reason, f"达到{reason.removeprefix('TARGET_')}分批保护位")})
                seen.add(reason)
        peak = max(peak, float(a["index_high"][j]))
    current = f.iloc[-1]
    scale = float(current.close / current.index_close)
    suspended = days[-1] != as_of
    status = "SELL" if pending else "REDUCE" if reduction else "SOFT_CONFIRMING" if breaches else "HOLD"
    reason = (XP.REASONS.get(pending) if pending else "分批保护信号已出现，尚未登记处置" if reduction
              else f"连续收盘失守 {breaches}/{required} 次" if breaches else "尚未触发持仓退出规则")
    return {**base, "status": "DATA_WAIT" if suspended else status,
            "reason": "已知停牌，等待可交易行情；此前风险仍需处置" if suspended else reason,
            "pending_action": status if suspended else "", "events": events,
            "return_pct": float((current.index_close / price - 1) * 100),
            "raw_return_pct": float((current.close / position["entry_price"] - 1) * 100),
            "hard_floor": hard * scale, "soft_floor": soft * scale, "soft_breaches": breaches,
            "soft_required": required, "held_sessions": len(f) - entry,
            "strong": bool(current.exit_strong), "market_known": bool(current.market_known),
            "pending_reason": pending, "alert_since": events[0]["date"] if events else ""}


def lad_review(f, entry, days, as_of, base, position=None, g1=frozenset()):
    """S4 staged profit-taking (LAD), advice only; mirrors research/exit_study_s4.py on the paper close entry:
    1/3 at +10% and 1/3 at +20% (intraday touch); the rest leaves at the next open after two closes < EMA20 x 0.97,
    or, once a close has been >= +10%, a heavy bearish candle or long upper shadow on >= 1.5x volume; -10% intraday
    stop; 60 sessions. Also reports the return a strict LAD execution would have made (adjusted prices, no fees).
    v2 graded denial (research/accept_failure.py B2h): within 3 sessions a close below the signal day's open sells
    half; a close below the pullback low (lowest low of the 10 sessions before the signal), or two closes below EMA20
    within 5 sessions, sells the rest. The signal day is the entry day (the paper entry is that day's close).
    v3: a session in g1 (the CSI 1000 trend exit, S4研究总结_20260925.md §4) sells the rest at the next open.
    The position's stored exit_policy_version selects the rules (LAD_POLICIES); older versions take G1 only from
    G1_MIGRATION. Stored s4_pullback_low / s4_signal_open (raw prices frozen at the pick) are used when present,
    converted to the adjusted series by the entry day's ratio; otherwise they are rebuilt from the frame.
    Every partial sale is capped at what is still held (research/accept_failure.py sell(): q = min(q, pos))."""
    position = position or {}
    version = position.get("exit_policy_version") or LAD_VERSION
    if version not in LAD_POLICIES:
        return {**base, "status": "DATA_WAIT", "reason": f"未知的分批止盈版本 {version}，不按现行规则替代回放"}
    policy = LAD_POLICIES[version]
    g1_from = "" if policy["g1"] else G1_MIGRATION
    o, h, l, c = (f[k].to_numpy(float) for k in ("index_open", "index_high", "index_low", "index_close"))
    ema20 = f.index_close.ewm(span=20, adjust=False).mean().to_numpy()
    vol = (f["volume"] if "volume" in f else f["amount"]).astype(float)
    vr = vol / vol.shift(1).rolling(20).mean()
    e = c[entry]
    held, realized, events = 1.0, 0.0, []
    pending, exited, below, armed, sold = "", None, 0, False, 0
    to_adj = float(f.index_close.iloc[entry] / f.close.iloc[entry])  # raw price -> adjusted series on the entry day
    stored = position.get("s4_pullback_low") is not None and position.get("s4_signal_open") is not None
    pb_low = float(position["s4_pullback_low"]) * to_adj if stored else l[max(0, entry - 10):entry].min()
    sig_open = float(position["s4_signal_open"]) * to_adj if stored else o[entry]
    half, warned, under = False, False, 0
    for j in range(entry + 1, len(f)):
        if pending:
            realized += held * o[j] / e
            held, exited = 0.0, (days[j], pending)
            break
        if half:
            realized += held / 2 * o[j] / e
            held, half = held / 2, False
        if l[j] <= e * 0.90:
            realized += held * min(o[j], e * 0.90) / e
            held, exited = 0.0, (days[j], "止损 −10%")
            events.append({"date": days[j], "reason": "LAD_STOP", "label": "盘中触及 −10% 止损，卖出剩余"})
            break
        for n, level in ((1, 1.10), (2, 1.20)):
            if sold < n and h[j] >= e * level:
                sold = n
                q = min(1 / 3, held)
                realized += q * max(o[j], e * level) / e
                held -= q
                events.append({"date": days[j], "reason": f"LAD_TP{n}",
                               "label": f"达到 +{round((level - 1) * 100)}%，按规则卖出 1/3" + ("（即剩余全部）" if held <= 1e-9 else "")})
        if held <= 1e-9:
            held, exited = 0.0, (days[j], "分批止盈后已无剩余")
            break
        armed = armed or c[j] >= e * 1.10
        below = below + 1 if c[j] < ema20[j] * 0.97 else 0
        under = under + 1 if j - entry <= 5 and c[j] < ema20[j] else 0
        if policy["b2h"] and not warned and j - entry <= 3 and c[j] < sig_open:
            warned = half = True
            events.append({"date": days[j], "reason": "LAD_WARN", "label": "恢复阳线被吞没（收盘低于信号日开盘），下一交易日开盘卖一半"})
        body = (o[j] - c[j]) / c[j - 1]
        rng = h[j] - l[j]
        heavy = (c[j] / c[j - 1] - 1 <= -0.05 or body >= 0.04) and vr.iloc[j] >= 1.5
        shadow = rng > 0 and (h[j] - max(o[j], c[j])) / rng >= 0.5 and rng / c[j - 1] >= 0.05 and vr.iloc[j] >= 1.5
        if days[j] in g1 and days[j] >= g1_from:
            pending = "大盘趋势清仓（G1：中证1000 收盘跌破 MA20 且 MA20 下拐）"
        elif policy["b2h"] and c[j] < pb_low:
            pending = "收盘跌破回踩低点（回踩事件被否定）"
        elif policy["b2h"] and under >= 2:
            pending = "买入后 5 日内两次收于 EMA20 下（恢复被否定）"
        elif below >= 2:
            pending = "两日收盘跌破 EMA20×0.97"
        elif armed and (heavy or shadow):
            pending = "放量大阴线" if heavy else "放量长上影"
        elif j - entry >= 60:
            pending = "持有满 60 个交易日"
        if pending:
            events.append({"date": days[j], "reason": "LAD_EXIT", "label": f"{pending}，下一交易日开盘卖出剩余"})
    scale = float(f.close.iloc[-1] / f.index_close.iloc[-1])
    mark = realized + held * c[-1] / e
    latest = [x for x in events if x["date"] == days[-1]]
    status = ("SELL" if exited or pending else
              "REDUCE" if any(x["reason"].startswith(("LAD_TP", "LAD_WARN")) for x in latest) else "HOLD")
    done = f"已按规则卖出 {1 - held:.0%}" if held < 1 else "尚未减仓"
    reason = (f"{exited[0]} 按规则卖出剩余（{exited[1]}）" if exited else f"{pending}：下一交易日开盘卖出剩余" if pending
              else "；".join(x["label"] for x in latest) if status == "REDUCE" else f"持有；{done}")
    suspended = days[-1] != as_of
    return {**base, "status": "DATA_WAIT" if suspended else status, "reason": reason, "events": events,
            "strategy": "S4_LAD", "policy_version": version,
            "g1_rule": "入场版本自带" if policy["g1"] else f"迁移：{G1_MIGRATION} 起适用（用户 2026-09-29 批准）",
            "anchor_source": "记入时冻结" if stored else "按行情重建",
            "return_pct": float((c[-1] / e - 1) * 100), "raw_return_pct": float((c[-1] / e - 1) * 100),
            "lad_return_pct": float((mark - 1) * 100), "lad_remaining": round(held, 4),
            "hard_floor": e * 0.90 * scale, "soft_floor": ema20[-1] * 0.97 * scale,
            "tp1": e * 1.10 * scale, "tp2": e * 1.20 * scale,
            "pullback_low": pb_low * scale, "signal_open": sig_open * scale,
            "held_sessions": len(f) - entry, "pending_reason": pending, "alert_since": events[0]["date"] if events else "",
            "pending_action": status if suspended else ""}


def scan_position(position, as_of):
    frame = load_frame(position["symbol"], as_of)
    result = evaluate(position, frame, as_of, market_calendar(as_of), XP.load_market(as_of), load_g1(as_of))
    if result["status"] != "DATA_WAIT":
        try:
            cycle, _ = SL.analyze(position["symbol"], as_of, trajectory=pd.DataFrame())
            result["lifecycle"] = cycle.get("cycle_stage_label", "未形成")
            result["cycle_id"] = cycle.get("cycle_id", "")
        except (OSError, ValueError, KeyError) as exc:
            result["lifecycle"] = f"生命周期待核验：{exc}"
    return result


def run(as_of, db=DB_PATH, output_root=None):
    checked_date(as_of)
    records = []
    for position in positions(db):
        if position["entry_date"] > as_of or (position["status"] == "CLOSED" and position["exit_date"] <= as_of):
            continue
        try:
            observation = scan_position(position, as_of)
        except (OSError, ValueError, KeyError) as exc:
            observation = {"as_of": as_of, "version": VERSION, "status": "DATA_WAIT", "reason": str(exc), "events": []}
        with closing(connect(db)) as conn, conn:
            conn.execute("INSERT OR REPLACE INTO observations VALUES (?, ?, ?)",
                         (position["id"], as_of, json.dumps(observation, ensure_ascii=False, allow_nan=False)))
        records.append({**position, "observation": observation})
        print(f"[virtual_portfolio] {position['symbol']} {LABELS[observation['status']]}: {observation['reason']}", flush=True)
    counts = {key: sum(r["observation"]["status"] == key for r in records) for key in LABELS}
    payload = {"as_of": as_of, "version": VERSION, "notice": NOTICE, "records": records,
               "summary": {"total": len(records), **counts, "status": "PARTIAL" if counts["DATA_WAIT"] else "SUCCESS"}}
    output = (SA.OUT_ROOT if output_root is None else Path(output_root)) / as_of.replace("-", "")
    _write(output / "virtual_positions.json", payload)
    lines = [f"# 虚拟持仓监控 {as_of}", "", NOTICE, "", f"持仓 {len(records)} | 卖出 {counts['SELL']} | 减仓 {counts['REDUCE']} | 待核验 {counts['DATA_WAIT']}", ""]
    for r in records:
        o = r["observation"]
        lines += [f"## {r['name']} {r['symbol']}", f"- 加入：{r['entry_date']} 收盘 {r['entry_price']:.3f}",
                  f"- 评估：{LABELS[o['status']]}；{o['reason']}",
                  f"- 生命周期：{o.get('lifecycle', '待核验')}；行情日期：{o.get('price_date', '未知')}",
                  f"- 复权价差：{o.get('return_pct', '--')}%；硬底线：{o.get('hard_floor', '--')}；收盘保护：{o.get('soft_floor', '--')}"]
        if o.get("strategy") == "S4_LAD":
            lines.append(f"- S4 分批止盈：剩余 {o['lad_remaining']:.0%}；止盈位 {o['tp1']:.2f} / {o['tp2']:.2f}；"
                         f"按规则执行的模拟收益 {o['lad_return_pct']:.1f}%")
        lines.append("")
    (output / "virtual_positions.md").write_text("\n".join(lines), encoding="utf-8")
    print(f"[virtual_portfolio] {as_of} total={len(records)} sell={counts['SELL']} reduce={counts['REDUCE']} data_wait={counts['DATA_WAIT']} -> {output / 'virtual_positions.md'}", flush=True)
    return payload


def listing(db=DB_PATH):
    rows = positions(db)
    if rows:
        with closing(connect(db)) as conn:
            for row in rows:
                found = conn.execute("SELECT payload FROM observations WHERE position_id=? ORDER BY as_of DESC LIMIT 1", (row["id"],)).fetchone()
                row["observation"] = json.loads(found[0]) if found else {
                    "status": "NEW", "reason": "等待日常持仓扫描", "events": [],
                    "price_date": row["entry_date"], "close": row["entry_price"], "return_pct": 0.,
                    "hard_floor": row["entry_price"] * (1 - 2 * row["atr_fraction"])}
                if row["observation"]["status"] == "DATA_WAIT":
                    for prior in conn.execute("SELECT payload FROM observations WHERE position_id=? ORDER BY as_of DESC", (row["id"],)):
                        alert = json.loads(prior[0])
                        if alert["status"] in {"SELL", "REDUCE"}:
                            row["last_actionable_observation"] = alert
                            break
    try:
        report_date = SA.latest_report_date()
    except FileNotFoundError:
        report_date = ""
    return {"positions": rows, "notice": NOTICE, "labels": LABELS, "version": VERSION, "report_date": report_date}


def close_position(position_id, as_of, db=DB_PATH):
    with closing(connect(db)) as conn:
        found = conn.execute("SELECT payload FROM positions WHERE id=?", (position_id,)).fetchone()
    if not found:
        raise ValueError("虚拟持仓不存在")
    position = json.loads(found[0])
    if position["status"] == "CLOSED":
        return position
    if as_of <= position["entry_date"]:
        raise ValueError("按A股T+1约束，加入当日不能模拟卖出")
    f, quote = close_quote(position["symbol"], as_of)
    from . import data as D
    limit = D.limit_pct(position["symbol"])
    if limit == 10 and quote.get("isST") == 1:
        limit = 5
    lower = math.floor(float(quote.preclose) * (1 - limit / 100) * 100 + .5) / 100
    if float(quote.high) <= lower + .001:
        raise ValueError("当日一字跌停，不能假设卖出成交")
    with closing(connect(db)) as conn, conn:
        conn.execute("BEGIN IMMEDIATE")
        latest = conn.execute("SELECT MAX(as_of) FROM observations WHERE position_id=?", (position_id,)).fetchone()[0]
        if latest and as_of < latest:
            raise Conflict("卖出日期早于已保存的持仓评估，请使用最新行情日")
        old = json.loads(conn.execute("SELECT payload FROM positions WHERE id=?", (position_id,)).fetchone()[0])
        if old["status"] == "CLOSED":
            return old
        entry = f[f.date.eq(position["entry_date"])]
        if entry.empty or not math.isclose(float(entry.close.iloc[0]), position["entry_price"], rel_tol=1e-6):
            raise ValueError("入场行情缺失或已修订，请核验后再卖出")
        position.update(status="CLOSED", exit_date=as_of, exit_price=float(quote.close),
                        closed_at=datetime.now(timezone.utc).isoformat(),
                        return_pct=float((quote.index_close / entry.index_close.iloc[0] - 1) * 100))
        conn.execute("UPDATE positions SET status='CLOSED', payload=? WHERE id=?", (json.dumps(position, ensure_ascii=False), position_id))
    return position
