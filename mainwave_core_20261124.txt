"""Paper-only order gateway. Integer cash; fills, not signals, change holdings."""
from contextlib import contextmanager
from datetime import datetime
from decimal import Decimal, ROUND_HALF_UP
import hashlib
import json
import math
import re
from . import store

VERSION = "paper_gateway_v1_20260919"
MODEL = "NEXT_OPEN_FIXED_CAP_PAPER_ONLY"
ACTIVE = ("OPEN", "PARTIAL", "UNKNOWN", "EXIT_BLOCKED")
NOTICE = "独立模拟账户，非真实成交。撮合假设为已提交订单后的交易日开盘、固定单次股数上限；涨跌停开盘不假定成交，不模拟真实排队。旧虚拟持仓不自动迁入。"


class Conflict(ValueError):
    pass


def cents(value):
    try:
        d = Decimal(str(value))
        if not d.is_finite():
            raise ValueError("金额必须有限")
        return int((d * 100).quantize(Decimal("1"), rounding=ROUND_HALF_UP))
    except Exception as exc:
        raise ValueError("金额无效") from exc


def fee(notional, cfg, side):
    commission = max(cfg["min_fee_cents"], math.ceil(notional * cfg["fee_bps"] / 10000))
    return commission + (math.ceil(notional * cfg["sell_tax_bps"] / 10000) if side == "SELL" else 0)


@contextmanager
def database(db=store.DB_PATH):
    with store.connect(db) as conn:
        # Individual statements preserve the encompassing immediate transaction.
        for sql in (
            "CREATE TABLE IF NOT EXISTS paper_account (id INTEGER PRIMARY KEY CHECK(id=1), initial_cents INTEGER NOT NULL, cash_cents INTEGER NOT NULL, created_at TEXT NOT NULL, config TEXT NOT NULL)",
            "CREATE TABLE IF NOT EXISTS paper_plans (id TEXT PRIMARY KEY, snapshot_id TEXT NOT NULL REFERENCES snapshots(id), opportunity_id TEXT NOT NULL, created_at TEXT NOT NULL, payload TEXT NOT NULL)",
            "CREATE TABLE IF NOT EXISTS paper_orders (id TEXT PRIMARY KEY, plan_id TEXT NOT NULL REFERENCES paper_plans(id), side TEXT NOT NULL, quantity INTEGER NOT NULL, filled INTEGER NOT NULL, status TEXT NOT NULL, reserve_cents INTEGER NOT NULL, risk_cents INTEGER NOT NULL, created_at TEXT NOT NULL, payload TEXT NOT NULL)",
            "CREATE TABLE IF NOT EXISTS paper_fills (id TEXT PRIMARY KEY, order_id TEXT NOT NULL REFERENCES paper_orders(id), day TEXT NOT NULL, quantity INTEGER NOT NULL, price_cents INTEGER NOT NULL, fee_cents INTEGER NOT NULL)",
            "CREATE TABLE IF NOT EXISTS paper_lots (id TEXT PRIMARY KEY REFERENCES paper_fills(id), plan_id TEXT NOT NULL REFERENCES paper_plans(id), symbol TEXT NOT NULL, entry_day TEXT NOT NULL, quantity INTEGER NOT NULL, remaining INTEGER NOT NULL, price_cents INTEGER NOT NULL, entry_fee_cents INTEGER NOT NULL)",
            "CREATE TABLE IF NOT EXISTS paper_events (id TEXT PRIMARY KEY, created_at TEXT NOT NULL, kind TEXT NOT NULL, payload TEXT NOT NULL)",
            "CREATE TABLE IF NOT EXISTS paper_valuations (id TEXT PRIMARY KEY, as_of TEXT NOT NULL, plan_id TEXT NOT NULL REFERENCES paper_plans(id), payload TEXT NOT NULL)",
        ):
            conn.execute(sql)
        yield conn


def event(conn, kind, body, key=None):
    identity = key or store.digest([kind, body])
    conn.execute("INSERT OR IGNORE INTO paper_events VALUES (?,?,?,?)", (identity, store.now(), kind, store.encoded(body)))


def initialize(body, db=store.DB_PATH, at=None):
    fields = {"capital", "fee_bps", "min_fee", "sell_tax_bps", "single_risk_pct", "total_risk_pct", "symbol_cap_pct", "theme_cap_pct", "gross_cap_pct", "max_fill_lots"}
    # Optional account protocol keys (a compounding account); absent keys keep the original behaviour exactly.
    optional = {"capital_policy", "equity_ceiling", "stock_seats"}
    if not isinstance(body, dict) or not fields <= set(body) <= fields | optional:
        raise ValueError("需明确初始本金、费率、单笔/总风险、个股/题材/总敞口及模拟单次成交手数")
    extra = {k: body[k] for k in optional if k in body}
    body = {k: v for k, v in body.items() if k in fields}
    if any(isinstance(v, bool) or not isinstance(v, (int, float, str)) for v in body.values()):
        raise ValueError("Account settings must be numeric")
    cfg = {k: float(v) for k, v in body.items()}
    policy = extra.get("capital_policy", "INITIAL_CAP")
    if policy not in ("INITIAL_CAP", "COMPOUND_CEILING"):
        raise ValueError("资本口径只能是 INITIAL_CAP 或 COMPOUND_CEILING")
    ceiling = float(extra.get("equity_ceiling", cfg["capital"]))
    seats = extra.get("stock_seats")
    if not math.isfinite(ceiling) or ceiling < cfg["capital"]:
        raise ValueError("扩容上限不得低于初始本金")
    if seats is not None and (isinstance(seats, bool) or not isinstance(seats, int) or not 1 <= seats <= 10):
        raise ValueError("股票席位需为1至10的整数")
    if not all(math.isfinite(v) for v in cfg.values()):
        raise ValueError("配置不能含空值或非有限数")
    if not 1000 <= cfg["capital"] <= 100000000 or not 0 < cfg["single_risk_pct"] <= cfg["total_risk_pct"] <= 20:
        raise ValueError("本金或风险范围无效")
    if not 0 < cfg["symbol_cap_pct"] <= cfg["theme_cap_pct"] <= cfg["gross_cap_pct"] <= 100:
        raise ValueError("敞口需满足 个股 <= 题材 <= 总敞口 <= 100%")
    if any(not 0 <= cfg[k] <= 100 for k in ("fee_bps", "sell_tax_bps", "min_fee")):
        raise ValueError("模拟费率/最低手续费超出允许范围")
    if not 1 <= cfg["max_fill_lots"] <= 1000 or cfg["max_fill_lots"] % 1:
        raise ValueError("模拟成交手数需为1至1000的整数")
    capital = cents(cfg.pop("capital"))
    cfg["min_fee_cents"] = cents(cfg.pop("min_fee"))
    cfg.update(model=MODEL, version=VERSION, first_fraction=.5, lot_size=100,
               additions_enabled=False, live_orders_enabled=False)
    if extra:
        cfg.update(capital_policy=policy, equity_ceiling_cents=cents(ceiling), stock_seats=seats)
    with database(db) as conn:
        if conn.execute("SELECT 1 FROM paper_account").fetchone():
            raise Conflict("模拟账户已初始化；不允许重置本金或覆盖历史账本")
        created = at or store.now()
        conn.execute("INSERT INTO paper_account VALUES (1,?,?,?,?)", (capital, capital, created, store.encoded(cfg)))
        event(conn, "ACCOUNT_INITIALIZED", {"capital_cents": capital, "config": cfg})
    return view(db)


def _account(conn):
    row = conn.execute("SELECT * FROM paper_account WHERE id=1").fetchone()
    if not row:
        raise Conflict("尚未配置模拟账户；目前只保存观察和排名")
    return dict(row), json.loads(row["config"])


def source_fingerprint():
    return {p.name: hashlib.sha256(p.read_bytes()).hexdigest() for p in
            (store.ROOT / "mainwave_engine/exit_policy.py", store.ROOT / "mainwave_engine/pullback_plan.py")}


def _exposure(conn):
    values = []
    for lot in conn.execute("SELECT l.*, p.payload FROM paper_lots l JOIN paper_plans p ON p.id=l.plan_id WHERE remaining>0"):
        p = json.loads(lot["payload"])
        valuation = conn.execute("SELECT payload FROM paper_valuations WHERE plan_id=? ORDER BY as_of DESC,rowid DESC LIMIT 1", (lot["plan_id"],)).fetchone()
        v = json.loads(valuation[0]) if valuation else {}
        mark = v.get("mark_cents", lot["price_cents"])
        values.append({"symbol": lot["symbol"], "theme": p["theme"],
                       "cost": lot["remaining"] * max(mark, lot["price_cents"]), "market_value": lot["remaining"] * mark,
                       "valuation_day": v.get("day"), "valuation_status": v.get("status"),
                       "risk": lot["remaining"] * max(0, mark - p["stop_cents"])})
    for o in conn.execute("SELECT o.*,p.payload AS plan FROM paper_orders o JOIN paper_plans p ON p.id=o.plan_id WHERE o.side='BUY' AND o.status IN ('OPEN','PARTIAL','UNKNOWN')"):
        p = json.loads(o["plan"])
        values.append({"symbol": p["symbol"], "theme": p["theme"], "cost": o["reserve_cents"], "risk": o["risk_cents"]})
    return values


def submit(identity, symbol, db=store.DB_PATH, at=None, book="MANUAL"):
    """Explicit approval from UI/CLI (MANUAL) or the frozen mechanical rule (AUTO); never both in one ledger."""
    if book not in {"MANUAL", "AUTO", "NEW100K"}:
        raise ValueError("账本只能是 MANUAL、AUTO 或 NEW100K")
    at = at or store.now()
    with database(db) as conn:
        account, cfg = _account(conn)
        snap = store.snapshot(conn, identity)
        current = store.latest(conn)
        if current["snapshot_id"] != identity:
            raise Conflict("快照已更新，请按最新证据重新复核")
        if snap["recorded_at"] > at or snap["as_of"] > at[:10] or (datetime.fromisoformat(at[:10]) - datetime.fromisoformat(snap["as_of"])).days > 7:
            raise Conflict("快照时间无效或过期")
        row = next((r for r in snap["records"] if r["symbol"] == symbol), None)
        if not row or not row["paper_plan_eligible"]:
            raise Conflict("当前机会未通过回踩、市场、数据与风险条件")
        scope = row.get("alpha_scope") or ranking_scope(row.get("route_ids", ""))
        if book in ("AUTO", "NEW100K") and scope == "R3_RESEARCH":
            raise Conflict("R3 仅为研究线索，不进入自动策略账")
        if not re.fullmatch(r"(?:sh_60|sz_00)\d{4}", symbol) or any(x in row["name"].upper() for x in ("ST", "退")):
            raise Conflict("首期模拟执行仅普通主板；其他板块继续观察")
        plan_id = store.digest([row["opportunity_id"], VERSION])[:24]
        if conn.execute("SELECT 1 FROM paper_plans WHERE id=?", (plan_id,)).fetchone():
            return {"plan_id": plan_id, "created": False}
        if any(e["symbol"] == symbol for e in _exposure(conn)):
            raise Conflict("该股票已有持仓或未决买单，不叠加另一模式预算")
        w = row["window"]
        low, limit, stop, atr = (cents(w.get(k)) for k in ("lower", "upper", "invalid", "frozen_atr"))
        if not 0 < stop < low <= limit or atr <= 0:
            raise ValueError("冻结结构边界或ATR无效")
        exposures = _exposure(conn)
        if any(e.get("market_value") is not None and (e.get("valuation_day") != snap["as_of"] or e.get("valuation_status") != "HOLD") for e in exposures):
            raise Conflict("已有持仓缺少同日有效估值，先复核持仓风险再开新仓")
        if conn.execute("SELECT 1 FROM paper_orders WHERE status='UNKNOWN' OR (side='SELL' AND status IN ('OPEN','PARTIAL','EXIT_BLOCKED'))").fetchone():
            raise Conflict("先处理未决订单和退出风险，再申请新仓")
        reserved = sum(r[0] for r in conn.execute("SELECT reserve_cents FROM paper_orders"))
        equity = account["cash_cents"] + sum(x.get("market_value", 0) for x in exposures)
        # INITIAL_CAP (original): gains never widen the base. COMPOUND_CEILING: losses shrink it at once, gains widen it
        # up to the approved ceiling, which only a reviewed protocol change raises.
        capital = (min(equity, cfg["equity_ceiling_cents"]) if cfg.get("capital_policy") == "COMPOUND_CEILING"
                   else min(account["initial_cents"], equity))
        seats = cfg.get("stock_seats")
        if seats and len({e["symbol"] for e in exposures}) >= seats:
            event(conn, "PLAN_REJECTED", {"symbol": symbol, "snapshot_id": identity, "reason": "NO_SEAT_STOCK"})
            return {"created": False, "reason": f"股票席位已满（{int(seats)} 只，含未决买单）"}
        total = sum(x["cost"] for x in exposures)
        theme = sum(x["cost"] for x in exposures if x["theme"] == row["theme"])
        budget = min(account["cash_cents"] - reserved, capital * cfg["symbol_cap_pct"] / 100 * cfg["first_fraction"],
                     capital * cfg["theme_cap_pct"] / 100 - theme, capital * cfg["gross_cap_pct"] / 100 - total)
        risk_budget = min(capital * cfg["single_risk_pct"] / 100,
                          capital * cfg["total_risk_pct"] / 100 - sum(x["risk"] for x in exposures))
        # Fixed hard floor may be below the structure; size using the worse boundary.
        hard = min(stop, low - 2 * atr)
        if hard <= 0:
            raise ValueError("风险边界无效")
        quantity = max(0, int(min(budget / limit, risk_budget / (limit - hard))) // 100 * 100)
        while quantity and (quantity * limit + fee(quantity * limit, cfg, "BUY") > budget or
                            quantity * (limit - hard) + fee(quantity * limit, cfg, "BUY") + fee(quantity * hard, cfg, "SELL") > risk_budget):
            quantity -= 100
        if quantity < 100:
            event(conn, "PLAN_REJECTED", {"symbol": symbol, "snapshot_id": identity, "reason": "BUDGET_BELOW_MINIMUM"})
            return {"created": False, "reason": "现金或风险预算不足最小申报量"}
        from .. import exit_policy as XP
        from . import rules as RL
        payload = {"id": plan_id, "snapshot_id": identity, "opportunity_id": row["opportunity_id"],
                   "symbol": symbol, "name": row["name"], "theme": row["theme"], "route_ids": row["route_ids"],
                   "book": book, "alpha_scope": scope, "strength_rank": row.get("strength_rank"),
                   "participation_rank": row.get("participation_rank"), "strength_score": row.get("strength_score"),
                   "mode": "TREND_PULLBACK", "setup_kind": row["setup_kind"], "entry_day": snap["as_of"],
                   "lower_cents": low, "limit_cents": limit, "stop_cents": hard, "structure_cents": stop,
                   "atr_cents": atr, "quantity": quantity, "version": VERSION, "created_at": at,
                   "knowledge_registry_id": snap["knowledge_registry_id"], "ranking_version": snap["version"],
                   "knowledge_ids": ["F06-K04", "F10-K04", "F02-K07"], "exit_version": XP.VERSION,
                   "exit_rules": XP.RULES, "exit_code_hashes": source_fingerprint(),
                   "first_fraction": .5, "additions_enabled": True, "model": MODEL,
                   "planned_budget_cents": quantity * limit * 2,
                   "rule_fingerprint": RL.fingerprint(),
                   "exit_execution": "收盘识别风险后下个可交易开盘执行；不是盘中止损成交；部分止盈也按下一开盘模拟。"}
        conn.execute("INSERT INTO paper_plans VALUES (?,?,?,?,?)", (plan_id, identity, row["opportunity_id"], at, store.encoded(payload)))
        oid = plan_id + ":initial"
        reserve = quantity * limit + fee(quantity * limit, cfg, "BUY")
        conn.execute("INSERT INTO paper_orders VALUES (?,?,?,?,?,?,?,?,?,?)",
                     (oid, plan_id, "BUY", quantity, 0, "OPEN", reserve,
                      quantity * (limit - hard) + fee(quantity * limit, cfg, "BUY") + fee(quantity * hard, cfg, "SELL"),
                      at, store.encoded({"reason": "INITIAL_HALF_BUDGET", "model": MODEL})))
        event(conn, "ORDER_APPROVED", {"order_id": oid, "plan": payload, "reserved_cents": reserve})
        return {"plan_id": plan_id, "order_id": oid, "created": True}


def ranking_scope(route_ids):
    from . import ranking
    return ranking.alpha_scope(route_ids)


def _cancel(conn, oid, reason):
    conn.execute("UPDATE paper_orders SET status='CANCELLED',reserve_cents=0,risk_cents=0 WHERE id=?", (oid,))
    event(conn, "ORDER_CANCELLED", {"order_id": oid, "reason": reason})


def cancel(oid, db=store.DB_PATH):
    with database(db) as conn:
        o = conn.execute("SELECT * FROM paper_orders WHERE id=?", (oid,)).fetchone()
        if not o or o["side"] != "BUY" or o["status"] not in ACTIVE or o["status"] == "UNKNOWN":
            raise Conflict("只能取消已知状态的未完成模拟买单；退出任务不可在此取消")
        _cancel(conn, oid, "USER_CANCELLED")
    return view(db)


def order_band(order_payload, plan):
    """A conditional add carries its own confirmed band and reference day; the initial uses the plan window."""
    return (order_payload.get("min_price_cents", plan["lower_cents"]),
            order_payload.get("max_price_cents", plan["limit_cents"]),
            order_payload.get("reference_day", plan["entry_day"]))


def apply_fill(conn, oid, fill_id, day, quantity, price):
    """Internal paper adapter only. Callbacks are validated and deduplicated transactionally."""
    old = conn.execute("SELECT * FROM paper_fills WHERE id=?", (fill_id,)).fetchone()
    price = cents(price)
    if old:
        if (old["order_id"], old["day"], old["quantity"], old["price_cents"]) != (oid, day, quantity, price):
            raise Conflict("相同成交ID内容冲突")
        return False
    account, cfg = _account(conn)
    o = conn.execute("SELECT * FROM paper_orders WHERE id=?", (oid,)).fetchone()
    if not o or o["status"] not in ACTIVE or isinstance(quantity, bool) or not isinstance(quantity, int) or quantity <= 0 or quantity > o["quantity"] - o["filled"] or price <= 0:
        raise ValueError("成交数量或订单状态无效")
    if day < o["created_at"][:10]:
        raise ValueError("成交不能早于委托")
    plan = json.loads(conn.execute("SELECT payload FROM paper_plans WHERE id=?", (o["plan_id"],)).fetchone()[0])
    notional = quantity * price
    charge = fee(notional, cfg, o["side"])
    if o["side"] == "BUY":
        low, high, _ = order_band(json.loads(o["payload"]), plan)
        if quantity % 100 or price > high or price < low:
            raise ValueError("买入成交违反冻结价格/数量限制")
        other_reserve = sum(r[0] for r in conn.execute("SELECT reserve_cents FROM paper_orders WHERE id<>?", (oid,)))
        if notional + charge > account["cash_cents"] - other_reserve:
            raise Conflict("成交超过可用资金")
        delta = -notional - charge
    else:
        lots = list(conn.execute("SELECT * FROM paper_lots WHERE plan_id=? AND remaining>0 AND entry_day<? ORDER BY entry_day,id", (o["plan_id"], day)))
        if sum(r["remaining"] for r in lots) < quantity:
            raise Conflict("可售不足或T+1限制")
        delta = notional - charge
        remaining = quantity
        for lot in lots:
            used = min(remaining, lot["remaining"])
            conn.execute("UPDATE paper_lots SET remaining=remaining-? WHERE id=?", (used, lot["id"]))
            remaining -= used
            if not remaining:
                break
    conn.execute("INSERT INTO paper_fills VALUES (?,?,?,?,?,?)", (fill_id, oid, day, quantity, price, charge))
    if o["side"] == "BUY":
        conn.execute("INSERT INTO paper_lots VALUES (?,?,?,?,?,?,?,?)",
                     (fill_id, o["plan_id"], plan["symbol"], day, quantity, quantity, price, charge))
    filled = o["filled"] + quantity
    left = o["quantity"] - filled
    cap = order_band(json.loads(o["payload"]), plan)[1] if o["side"] == "BUY" else plan["limit_cents"]
    reserve = left * cap + fee(left * cap, cfg, "BUY") if left and o["side"] == "BUY" else 0
    risk = left * (cap - plan["stop_cents"]) + fee(left * cap, cfg, "BUY") + fee(left * plan["stop_cents"], cfg, "SELL") if reserve else 0
    if reserve > account["cash_cents"] + delta - (other_reserve if o["side"] == "BUY" else 0):
        raise Conflict("部分成交后的最低手续费导致剩余预留不足")
    conn.execute("UPDATE paper_account SET cash_cents=cash_cents+? WHERE id=1", (delta,))
    conn.execute("UPDATE paper_orders SET filled=?,status=?,reserve_cents=?,risk_cents=? WHERE id=?",
                 (filled, "FILLED" if left == 0 else "PARTIAL", reserve, risk, oid))
    event(conn, "FILL", {"fill_id": fill_id, "order_id": oid, "day": day, "quantity": quantity, "price_cents": price, "fee_cents": charge})
    return True


def queue_exit(conn, plan_id, day, reason, fraction=1.):
    existing = conn.execute("SELECT * FROM paper_orders WHERE plan_id=? AND side='SELL' AND status IN ('OPEN','PARTIAL','UNKNOWN','EXIT_BLOCKED')", (plan_id,)).fetchall()
    lots = list(conn.execute("SELECT remaining FROM paper_lots WHERE plan_id=?", (plan_id,)))
    total = sum(r[0] for r in lots)
    if existing:
        if fraction < 1 or any(x["status"] == "UNKNOWN" for x in existing) or sum(x["quantity"] - x["filled"] for x in existing) >= total:
            return
        for o in existing:
            _cancel(conn, o["id"], "ESCALATE_TO_FULL_EXIT")
    qty = total
    if fraction < 1:
        qty = int(qty * fraction) // 100 * 100
    if qty <= 0:
        return
    oid = f"{plan_id}:exit:{day}:{reason}"
    if conn.execute("SELECT 1 FROM paper_orders WHERE id=?", (oid,)).fetchone():
        return
    conn.execute("INSERT INTO paper_orders VALUES (?,?,?,?,?,?,?,?,?,?)",
                 (oid, plan_id, "SELL", qty, 0, "OPEN", 0, 0, day + "T15:00:00+08:00", store.encoded({"reason": reason})))
    event(conn, "EXIT_REQUIRED", {"order_id": oid, "reason": reason, "quantity": qty})


def process_open(conn, day, bars):
    _, cfg = _account(conn)
    for o in conn.execute("SELECT o.*,p.payload AS plan FROM paper_orders o JOIN paper_plans p ON p.id=o.plan_id WHERE o.status IN ('OPEN','PARTIAL','EXIT_BLOCKED') ORDER BY o.side DESC,o.created_at,o.id").fetchall():
        if o["created_at"] >= day + "T09:30:00+08:00":
            continue
        fill_id = store.digest([o["id"], day, MODEL])
        if conn.execute("SELECT 1 FROM paper_fills WHERE id=?", (fill_id,)).fetchone():
            continue
        p = json.loads(o["plan"])
        bar = bars.get(p["symbol"], {})
        if p["exit_code_hashes"] != source_fingerprint() and o["side"] == "BUY":
            _cancel(conn, o["id"], "FROZEN_RULE_MISMATCH")
            continue
        if bar.get("date") != day or not bar.get("raw_verified"):
            if o["side"] == "SELL":
                conn.execute("UPDATE paper_orders SET status='EXIT_BLOCKED' WHERE id=?", (o["id"],))
            event(conn, "DATA_WAIT", {"order_id": o["id"], "day": day, "reason": "NO_VERIFIED_BAR"})
            continue
        op, pre = cents(bar["open"]), cents(bar["preclose"])
        # First deployment is ordinary main-board paper only. Unknown security status blocks matching.
        blocked = not bar.get("ordinary_mainboard") or bar.get("corporate_action") or not pre > 0
        at_limit = op >= round(pre * 1.1) - 1 if o["side"] == "BUY" else op <= round(pre * .9) + 1
        if blocked or at_limit:
            if o["side"] == "BUY":
                _cancel(conn, o["id"], "SECURITY_OR_LIMIT_NOT_FILLABLE")
            else:
                conn.execute("UPDATE paper_orders SET status='EXIT_BLOCKED' WHERE id=?", (o["id"],))
                event(conn, "EXIT_BLOCKED", {"order_id": o["id"], "day": day, "reason": "SECURITY_OR_LIMIT"})
            continue
        low, high, reference = order_band(json.loads(o["payload"]), p)
        if o["side"] == "BUY" and (paper_day_gap(o["created_at"], day) > 7 or bar.get("previous_market_day") != reference):
            _cancel(conn, o["id"], "MISSED_NEXT_SESSION_OR_EXPIRED")
            continue
        if o["side"] == "BUY" and not low <= op <= high:
            _cancel(conn, o["id"], "OPEN_OUTSIDE_FROZEN_WINDOW")
            continue
        qty = min(o["quantity"] - o["filled"], int(cfg["max_fill_lots"]) * 100)

        return result

def features(frame, day):
    f = frame[frame.date.le(day)].tail(80).copy()
    if len(f) < RULES["history_sessions"] or f.date.iloc[-1] != day:
        return {"status": "DATA_WAIT"}
    pct = pd.to_numeric(f.pct_chg, errors="coerce")
    close = pd.to_numeric(f.close, errors="coerce")

    if not all(math.isfinite(x) for x in pct) or pct.le(-100).any() or not all(math.isfinite(x) for x in close) or close.le(0).any() or volume is None or volume <= 0:
        return {"status": "DATA_WAIT"}

    movement = index.diff().abs().tail(10).sum()
    turnover = pd.to_numeric(f.turnover, errors="coerce") if "turnover" in f else pd.Series(dtype=float)
    base = turnover.iloc[-21:-1].median() if len(turnover) else None
    return store.clean({"status": "READY", "close": float(close.iloc[-1]),

        "close_location": float((latest.close - latest.low) / width) if width and width > 0 else None,

def theme_cohorts(records):
    """Reading of the candidate cohort itself: are the candidates in this theme still being pushed today?

    The cohort is already a filtered, self-referential sample; the result is named STRONG/WEAK for the cohort
    and never presented as the market's main line. Conditions are checked first, the top-N cut comes last."""
    groups = {}
    for r in records:
        f = r.get("features") or {}
        if f.get("status") != "READY" or r.get("theme") in (None, "", "未归属"):
            continue
        groups.setdefault(r["theme"], []).append(r)
    out = {}
    for theme, rows in groups.items():
        ex5 = [v for v in (number(x["features"].get("market_excess5")) for x in rows) if v is not None]
        strong = sum(1 for x in rows if trigger_of(x["features"]) == "ABSORBED")
        weak = sum(1 for x in rows if trigger_of(x["features"]) == "WEAK_ACTIVE")
        pushing = sum(1 for x in rows if stage_of(x)[0] in ("EARLY", "CONTINUATION"))
        n = len(rows)
        known = n >= THRESHOLDS["cohort_min_members"] and bool(ex5)
        med = round(statistics.median(ex5), 2) if ex5 else None
        out[theme] = {"state": "UNKNOWN" if not known else "NEUTRAL", "members": n, "median_excess5": med,
                      "strong_close": strong, "weak_active": weak, "pushing": pushing,
                      "qualifies_strong": known and med > 0 and strong >= 2 and pushing >= 2,
                      "qualifies_weak": known and med < 0 and weak * 2 >= n,
                      "basis": "CANDIDATE_COHORT_ONLY_NOT_MARKET_BREADTH"}
    strong_pool = sorted((t for t, c in out.items() if c["qualifies_strong"]), key=lambda t: -out[t]["median_excess5"])
    for theme in strong_pool[:THRESHOLDS["cohort_top"]]:
        out[theme]["state"] = "STRONG"
    weak_pool = sorted((t for t, c in out.items() if c["qualifies_weak"] and c["state"] != "STRONG"), key=lambda t: out[t]["median_excess5"])
    for theme in weak_pool[:THRESHOLDS["cohort_top"]]:
        out[theme]["state"] = "WEAK"
    return out


def market_theme_context(theme_facts):
    """Full-membership theme facts (eod_review) by theme name: breadth, limit-ups, table ranks. Dated by the facts."""
    if not theme_facts:
        return {}
    out = {}
    for key in ("top_by_limit_up", "top_by_limit_share", "top_by_amount_share", "chains"):
        for n, t in enumerate(theme_facts.get(key) or [], 1):
            name = t.get("theme")
            if not name:
                continue
            entry = out.setdefault(name, {"members": t.get("members"), "limit_up": t.get("limit_up"), "up": t.get("up"),
                                          "breadth": (t.get("breadth") or {}).get("value"), "ranks": {}, "as_of": theme_facts.get("as_of")})
            entry["ranks"][key] = n
    top = THRESHOLDS["market_theme_top"]
    for name, e in out.items():
        breadth = e["breadth"] if e["breadth"] is not None else 0.0
        listed_top = any(rank <= top for k, rank in e["ranks"].items() if k in ("top_by_limit_up", "top_by_limit_share"))
        e["state"] = ("STRONG" if listed_top and breadth >= THRESHOLDS["market_breadth_strong"]

def trigger_of(f):
    close_location, rel = number(f.get("close_location")), number(f.get("relative_turnover"))
    if close_location is None or rel is None:
        return "NEUTRAL"
    if close_location >= THRESHOLDS["strong_close"] and rel >= THRESHOLDS["absorb_turnover"]:
        return "ABSORBED"
    if close_location <= THRESHOLDS["weak_close"] and rel >= THRESHOLDS["weak_active_turnover"]:
        return "WEAK_ACTIVE"
    return "NEUTRAL"

    if stage in CONTINUATION:
        # A confirmed or forming first pullback right after the start still counts as early participation.
        if sv.get("setup_status") in ("CONFIRMED", "FORMING") and (record.get("source_record") or {}).get("local_role") in ("FIRST_RETEST", "ORIGIN_BREAKOUT"):
            return "EARLY", stage
        return "CONTINUATION", stage
    if stage in LATE:
        return "LATE", stage
    if stage in REPAIRING:
        return "REPAIRING", stage
    if stage in FAILED:
        return "FAILED", stage
    return "UNKNOWN", stage


def route_of(record):
    """Ordered 'prepared then ignited' only when the preparation date is known to precede the attempt date."""
    routes = {x for x in str(record.get("route_ids") or "").split("|") if x}
    re_ = (record.get("evidence") or {}).get("route_evidence") or {}
    if "R1" in routes and ("R2" in routes or re_.get("prepared_before_attempt")):
        prepared_at = re_.get("prepared_at") or re_.get("preparation_first_seen") or ""
        attempt_at = re_.get("attempt_at") or ""
        if re_.get("prepared_before_attempt") is True or (prepared_at and attempt_at and prepared_at < attempt_at):
            return "R2_THEN_R1"
        return "R1_AND_R2_UNORDERED"
    for r in ("R1", "R2", "R3"):
        if r in routes:
            return r
    return "NONE"


def preparation_of(record):
    re_ = (record.get("evidence") or {}).get("route_evidence") or {}
    state = re_.get("preparation_state") or (record.get("source_record") or {}).get("preparation_state") or ""
    return {"state": state, "live": state in LIVE_PREPARATION, "demand": bool(re_.get("demand")),
            "activity_contraction": bool(re_.get("activity_contraction")), "age": number(re_.get("preparation_age")),
            "prepared_at": re_.get("prepared_at") or re_.get("preparation_first_seen") or "", "attempt_at": re_.get("attempt_at") or ""}


def stage_transition(before, now):
    """Lifecycle movement is not a strength order: EARLY→CONTINUATION is progress, anything into LATE/FAILED is weakening,
    out of FAILED/REPAIRING into EARLY is a new event."""
    if before == now:
        return "UNCHANGED"
    if now in ("LATE", "FAILED"):
        return "WEAKENING"
    if before in ("FAILED", "REPAIRING") and now in ("EARLY", "CONTINUATION"):
        return "NEW"
    if before == "EARLY" and now == "CONTINUATION":
        return "PROGRESSING"
    if before in ("LATE", "REPAIRING") and now in ("EARLY", "CONTINUATION"):
        return "IMPROVING"
    if before == "CONTINUATION" and now == "EARLY":
        return "IMPROVING"
    return "UNCHANGED"


def change_of(current, previous):
    """Change from structured fields of the previous shadow row; without one, from the engine's evidence_change only
    when the sentence is unambiguous (a negated '改善' is not an improvement)."""
    if previous and previous.get("stage_bucket"):
        moves = [stage_transition(previous["stage_bucket"], current["stage_bucket"])]
        if previous.get("position") != current["position"]:
            moves.append("WEAKENING" if current["position"] == "EXTENDED" else "IMPROVING" if current["position"] == "IN_BAND" else "UNCHANGED")
        if previous.get("trigger") != current["trigger"]:
            moves.append("IMPROVING" if current["trigger"] == "ABSORBED" else "WEAKENING" if current["trigger"] == "WEAK_ACTIVE" else "UNCHANGED")
        prev_live = (previous.get("preparation") or {}).get("live")
        if prev_live is not None and prev_live != current["preparation"]["live"]:
            moves.append("IMPROVING" if current["preparation"]["live"] else "WEAKENING")
        if "NEW" in moves:
            return "NEW"
        if "WEAKENING" in moves:
            return "WEAKENING"
        if "IMPROVING" in moves:
            return "IMPROVING"
        if "PROGRESSING" in moves:
            return "PROGRESSING"
        return "UNCHANGED"
    text = str((current.get("_source") or {}).get("evidence_change") or "")
    if not text:
        return "UNKNOWN"
    if "新周期" in text or "首份" in text:
        return "NEW"
    negated = any(w in text for w in ("未改善", "无改善", "没有改善", "不改善"))
    weak = any(w in text for w in ("减弱", "消退", "退化", "恶化", "失效"))
    better = "改善" in text and not negated
    if better and not weak:
        return "IMPROVING"
    if weak and not better:
        return "WEAKENING"
    return "UNCHANGED" if ("延续" in text or "无新增" in text) else "UNKNOWN"


def re_eligible(record, as_of):
    """A continued-tracking name competes again only when its price and evidence carry the batch date and a structure
    is live. Re-entry to observation is not a trading permission; the frozen blockers except TRACKING_ONLY stay."""
    if not record.get("tracking_only") or not as_of:
        return False
    f = record.get("features") or {}
    sv = (record.get("evidence") or {}).get("state_vector") or {}
    src = record.get("source_record") or {}
    fresh = f.get("status") == "READY" and record.get("price_date") == as_of and (record.get("evidence") or {}).get("as_of") == as_of
    live = sv.get("setup_status") in ("CONFIRMED", "FORMING") or src.get("route_status") in ("CONFIRMATION_REVIEW", "PREPARATION_WATCH")
    return bool(fresh and live and "STRUCTURE_RISK" not in record.get("blockers", []))


def state_without(record, dropped):
    blockers = [b for b in record.get("blockers", []) if b not in dropped]
    if "DATA_WAIT" in blockers or "BENCHMARK_MISSING" in blockers:
        return "DATA_WAIT", blockers

def mainline_key(row):
    """Market breadth decides when a same-day table exists for the theme; otherwise the cohort reading."""
    if row.get("market_theme") in ("STRONG", "WEAK"):
        return MAINLINE_ORDER[row["market_theme"]]
    return MAINLINE_ORDER.get(row["mainline"], 1)


def keys(row):
    """Lexicographic keys for each ablation step; lower sorts first. No weights, no blended score. The order is a
    policy: state gate → position (extension is never outranked) → stage → theme → route → preparation → change →
    trigger → trend as the tie-break."""
    trend = -(row["trend"]["market_excess20"] if row["trend"]["market_excess20"] is not None else -999)
    frozen = (row["frozen_participation_rank"] if row["frozen_participation_rank"] is not None else 10 ** 6)
    base = (STATE_ORDER.get(row["state"], 9), row["alpha_scope"] == "R3_RESEARCH")
    position = (POSITION_ORDER[row["position"]],)
    trigger = (TRIGGER_ORDER[row["trigger"]],)
    stage_route = (STAGE_ORDER[row["stage_bucket"]], mainline_key(row), ROUTE_ORDER[row["route"]], 0 if row["preparation"]["live"] else 1)
    return {"V0_frozen": (frozen, row["symbol"]),
            "V1_tracking": (*base, frozen, row["symbol"]),
            "V2_position": (*base, *position, *trigger, trend, row["symbol"]),
            "V3_stage_route": (*base, *position, *stage_route, *trigger, trend, row["symbol"]),
            "V4_multiday": (*base, *position, *stage_route, CHANGE_ORDER[row["change"]], *trigger, trend, row["symbol"])}


def select_review_shortlist(rows, limit=2):
    """Names worth the next unit of review attention: at most `limit`, one per security, distinct themes,
    never extended, late or failed, never R3-only. It is not an allocation: it checks no holdings, orders,
    cash, ETF seat or account, and it grants nothing (see trading.allocation.allocate_ready_plans)."""
    chosen, seen = [], set()
    for row in rows:
        if len(chosen) >= limit:
            break
        if row["state"] not in ("PLAN_REVIEW", "WAIT_SETUP") or row["symbol"] in seen:
            continue
        if row["position"] != "IN_BAND" or row["stage_bucket"] in ("LATE", "FAILED", "REPAIRING") or row["alpha_scope"] == "R3_RESEARCH":
            continue
        if any(c["theme"] == row["theme"] and row["theme"] not in ("", "未归属") for c in chosen):
            continue
        chosen.append(row)
        seen.add(row["symbol"])
    return chosen


def watch_reconciliation(rows, preparation_watch):
    """Every count before any truncation: candidates, watch, overlap, outside, union."""
    known = {r["symbol"] for r in rows}
    watch_symbols = [w.get("symbol") for w in (preparation_watch or []) if w.get("symbol")]
    outside = [w for w in (preparation_watch or []) if w.get("symbol") and w["symbol"] not in known]
    return {"candidates": len(known), "watch": len(set(watch_symbols)), "overlap": len(known & set(watch_symbols)),
            "outside": len({w["symbol"] for w in outside}), "union": len(known | set(watch_symbols)),
            "note": "低偏离准备观察（影子）：域外名单进入待取证机会层，不进候选状态，无冻结触发价与失效价"}, outside

def risk_base(current_verified_equity, approved_equity_ceiling):
    """Fixed risk percentages apply to the smaller of verified equity and the approved ceiling: losses shrink the base
    at once, gains widen it only after the ceiling is raised by review."""
    if current_verified_equity is None:
        return None
    return min(current_verified_equity, approved_equity_ceiling) if approved_equity_ceiling else current_verified_equity


def _is_etf(symbol):
    return str(symbol).startswith(("sh_51", "sh_56", "sh_58", "sz_15", "sz_16"))


def plan_valid(plan, target_date):
    """Approved, has every required field, and its effective window covers the target session."""
    required = ("plan_id", "account_id", "symbol", "approval_state", "effective_trade_date", "entry_min", "entry_max", "invalidation", "quantity_cap", "planned_loss_budget")
    missing = [k for k in required if plan.get(k) in (None, "")]
    if missing:
        return False, "MISSING:" + ",".join(missing)
    if plan["approval_state"] != "APPROVED":
        return False, f"NOT_APPROVED:{plan['approval_state']}"
    if plan["effective_trade_date"] != target_date:
        return False, "EXPIRED_OR_NOT_YET_EFFECTIVE" if plan["effective_trade_date"] < target_date else "NOT_YET_EFFECTIVE"
    if not (plan["invalidation"] < plan["entry_min"] <= plan["entry_max"]):
        return False, "PRICE_BAND_INVALID"
    return True, ""


def allocate_ready_plans(plans, account, target_date, holdings=None, pending_orders=None, cash=None, verified_equity=None):
    """Pure decision: for each plan, allocate or name the block. Never creates or touches an order."""
    holdings = holdings or []
    pending = pending_orders or []
    account_id = account.get("account_id")
    result = {"version": VERSION, "account_id": account_id, "target_date": target_date, "allocated": [], "blocked": []}
    if account.get("status") != "ACTIVE" or any(v is None for v in (account.get("risk_profile") or {}).values()):
        for p in plans:
            result["blocked"].append({"plan_id": p.get("plan_id"), "symbol": p.get("symbol"), "reason": CONFIG_PENDING})
        result["note"] = "账户协议未确认：不分配任何计划"
        return result
    seats = account.get("seats") or {"stock": 2, "etf": 1}
    occupied_stock = {h["symbol"] for h in holdings if not _is_etf(h["symbol"])} | {o["symbol"] for o in pending if o.get("side", "BUY") == "BUY" and not _is_etf(o["symbol"])}
    occupied_etf = {h["symbol"] for h in holdings if _is_etf(h["symbol"])} | {o["symbol"] for o in pending if o.get("side", "BUY") == "BUY" and _is_etf(o["symbol"])}
    base = risk_base(verified_equity, account.get("approved_equity_ceiling"))
    profile = account["risk_profile"]
    reserved = sum(float(o.get("reserve", 0) or 0) for o in pending)
    available_cash = (float(cash) - reserved) if cash is not None else None
    used_risk = sum(float(h.get("risk", 0) or 0) for h in holdings) + sum(float(o.get("risk", 0) or 0) for o in pending)
    theme_used = {}
    for h in holdings:
        theme_used[h.get("theme")] = theme_used.get(h.get("theme"), 0.0) + float(h.get("cost", 0) or 0)
    for p in sorted(plans, key=lambda x: (x.get("priority", 10 ** 6), x.get("symbol", ""))):
        ok, reason = plan_valid(p, target_date)
        if not ok:
            result["blocked"].append({"plan_id": p.get("plan_id"), "symbol": p.get("symbol"), "reason": reason})
            continue
        if p["account_id"] != account_id:
            result["blocked"].append({"plan_id": p["plan_id"], "symbol": p["symbol"], "reason": f"ACCOUNT_MISMATCH:{p['account_id']}"})
            continue
        sym = p["symbol"]
        if sym in occupied_stock or sym in occupied_etf or any(a["symbol"] == sym for a in result["allocated"]):
            result["blocked"].append({"plan_id": p["plan_id"], "symbol": sym, "reason": "SECURITY_ALREADY_OCCUPIES_SEAT"})
            continue
        is_etf = _is_etf(sym)
        seat_used = len(occupied_etf) + sum(_is_etf(a["symbol"]) for a in result["allocated"]) if is_etf else len(occupied_stock) + sum(not _is_etf(a["symbol"]) for a in result["allocated"])
        if seat_used >= seats["etf" if is_etf else "stock"]:
            result["blocked"].append({"plan_id": p["plan_id"], "symbol": sym, "reason": "NO_SEAT_ETF" if is_etf else "NO_SEAT_STOCK"})
            continue
        if base is None or available_cash is None:
            result["blocked"].append({"plan_id": p["plan_id"], "symbol": sym, "reason": "EQUITY_OR_CASH_UNVERIFIED"})
            continue
        per_share_loss = p["entry_max"] - p["invalidation"]
        risk_room = min(base * profile["single_risk_pct"] / 100, base * profile["total_risk_pct"] / 100 - used_risk)
        cap_pct = profile["etf_cap_pct"] if is_etf else profile["symbol_cap_pct"]
        theme_room = base * profile["theme_cap_pct"] / 100 - theme_used.get(p.get("primary_theme"), 0.0)
        notional_room = min(available_cash, base * cap_pct / 100, theme_room if not is_etf else base * cap_pct / 100)
        qty = int(min(notional_room / p["entry_max"], risk_room / per_share_loss, p["quantity_cap"]) // 100 * 100) if per_share_loss > 0 else 0
        if qty < 100:
            result["blocked"].append({"plan_id": p["plan_id"], "symbol": sym, "reason": "CAPITAL_BLOCK",
                                      "detail": {"risk_room": round(risk_room, 2), "notional_room": round(notional_room, 2), "per_share_loss": round(per_share_loss, 4)}})
            continue
        planned_loss = qty * per_share_loss
        result["allocated"].append({"plan_id": p["plan_id"], "symbol": sym, "quantity": qty, "notional_max": round(qty * p["entry_max"], 2),
                                    "planned_loss": round(planned_loss, 2), "risk_base": round(base, 2), "theme": p.get("primary_theme")})
        available_cash -= qty * p["entry_max"]
        used_risk += planned_loss
        theme_used[p.get("primary_theme")] = theme_used.get(p.get("primary_theme"), 0.0) + qty * p["entry_max"]
    result["seats"] = {"stock_occupied_before": len(occupied_stock), "etf_occupied_before": len(occupied_etf), "limits": seats}
    return result

def _table(conn):
    conn.execute("CREATE TABLE IF NOT EXISTS rule_versions ("
                 "id TEXT PRIMARY KEY, kind TEXT NOT NULL, version TEXT NOT NULL, "
                 "created_at TEXT NOT NULL, payload TEXT NOT NULL)")


def _hash(path, root=None):
    target = Path(root or store.ROOT) / path
    return hashlib.sha256(target.read_bytes()).hexdigest() if target.exists() else ""


def build(root=None):
    """Current rule set: versions, parameters and direct file hashes in one summary."""
    from .. import entry_policy as EP, exit_policy as XP, pullback_plan as PP
    from . import modes, paper
    from . import VERSION as SOFTWARE
    parts = {

                  "parameters": {"initial": EP.INITIAL, "add_weights": list(EP.ADD_WEIGHTS),
                                 "pullbacks": list(EP.PULLBACKS), "window": EP.WINDOW}},
        "exit": {"version": XP.VERSION, "rules": XP.RULES, "parameters": {}},
        "pullback": {"version": getattr(PP, "VERSION", ""), "rules": {}, "parameters": {}},
        "mode": {"version": modes.VERSION, "rules": {}, "parameters": modes.CONTRACT_SUMMARY},
        "execution": {"version": paper.VERSION, "rules": {}, "parameters": {"model": paper.MODEL}},
        "dependency": {"version": _dependency_version(), "rules": {}, "parameters": {}},
    }
    for kind, item in parts.items():
        item["source"] = SOURCES[kind]
        item["sha256"] = _hash(SOURCES[kind], root)
    payload = {"registry_version": VERSION, "software_version": SOFTWARE, "parts": parts,
               "restorable": RESTORABLE, "freeze": FREEZE,
               "note": "软件版本号变化不等于策略冻结；策略、数据与研究协议版本分别记录。"}

def register(db=store.DB_PATH, root=None):
    """Record the current rule set so a later plan can look its own version back up."""
    payload = build(root)
    with store.connect(db) as conn:
        _table(conn)
        for kind, item in payload["parts"].items():
            identity = f"{kind}:{item['version']}:{item['sha256'][:12]}"
            conn.execute("INSERT OR IGNORE INTO rule_versions VALUES (?,?,?,?,?)",
                         (identity, kind, item["version"], store.now(), store.encoded(item)))
    return payload


def lookup(kind, version, sha256, db=store.DB_PATH, conn=None):
    """Inside a ledger transaction the caller's connection is reused; a second writer would self-lock."""
    if conn is not None:
        return _lookup(conn, kind, version, sha256)
    with store.connect(db) as own:
        return _lookup(own, kind, version, sha256)


def _lookup(conn, kind, version, sha256):
    _table(conn)
    row = conn.execute("SELECT payload FROM rule_versions WHERE id=?",
                       (f"{kind}:{version}:{sha256[:12]}",)).fetchone()
    if row:
        return json.loads(row[0])
    row = conn.execute("SELECT payload FROM rule_versions WHERE kind=? AND version=? ORDER BY rowid DESC LIMIT 1",
                       (kind, version)).fetchone()
    return json.loads(row[0]) if row else None


def dispatch(plan, db=store.DB_PATH, root=None, conn=None):
    """Decide which rules an existing plan runs under, and say so out loud."""
    current = build(root)
    frozen = plan.get("rule_fingerprint") or {}
    legacy = plan.get("exit_code_hashes") or {}
    parts, drift, unrestorable = {}, [], []
    for kind, item in current["parts"].items():
        want_version = frozen.get(kind, {}).get("version") or _legacy_version(plan, kind)
        want_hash = frozen.get(kind, {}).get("sha256") or _legacy_hash(legacy, kind)
        if not want_version and not want_hash:
            parts[kind] = {**item, "dispatch": "CURRENT_NO_FROZEN_RECORD"}
            continue
        same = (want_version in ("", item["version"])) and (want_hash in ("", item["sha256"]))
        if same:
            parts[kind] = {**item, "dispatch": "CURRENT_MATCHES_FROZEN"}
            continue
        drift.append(kind)
        registered = lookup(kind, want_version, want_hash or "", db, conn)
        if registered and registered.get("sha256") == want_hash:
            parts[kind] = {**registered, "dispatch": "FROZEN_PARAMETERS_FROM_REGISTRY",
                           "limit": RESTORABLE}
        else:
            unrestorable.append(kind)
            parts[kind] = {"version": want_version, "sha256": want_hash,
                           "dispatch": "UNRESTORABLE_PENDING_REVIEW", "limit": RESTORABLE}
    # Parameters can be read back from the registry; the code that ran them cannot. So any drift
    # means the frozen version is not executable: no new adds, deterministic risk floors keep running.
    status = ("PENDING_REVIEW" if unrestorable else "PARAMETERS_ON_RECORD_CODE_CHANGED" if drift else "CURRENT")
    return {"version": VERSION, "plan_id": plan.get("id"), "status": status, "parts": parts,
            "drift": drift, "unrestorable": unrestorable,
            "executable_as_frozen": not drift, "new_adds_allowed": not drift,
            "risk_tasks_retained": True,
            "note": ("规则已变更且登记表没有原版本，该计划待人工核对；不补仓，已有持仓只跑确定性硬底线与结构失效。"
                     if unrestorable else
                     "规则已变更：原参数可从登记表取回，但原代码不可重建，不算按冻结版本执行；"
                     "不补仓，已有持仓只跑确定性硬底线与结构失效，其余等待复核。"
                     if drift else "规则未变更，按当前版本运行。")}


def _legacy_version(plan, kind):
    return {"exit": plan.get("exit_version"), "entry": plan.get("entry_version"),
            "execution": plan.get("version")}.get(kind, "")


def _legacy_hash(legacy, kind):
    return {"exit": legacy.get("exit_policy.py"), "pullback": legacy.get("pullback_plan.py")}.get(kind, "")

    """Compact per-part version+hash record stored on every new plan."""
    current = build(root)
    return {kind: {"version": item["version"], "sha256": item["sha256"], "source": item["source"]}
            for kind, item in current["parts"].items()}

def signals(f):
    """S4 flags on an adjusted frame (open/high/low/close/volume). Returns a DataFrame aligned to f."""
    c, h, l = (f[k].to_numpy(float) for k in ("close", "high", "low"))
    s = pd.Series(c)
    ma5, ma20, ma60 = (s.rolling(n).mean().to_numpy() for n in (5, 20, 60))
    ema20 = s.ewm(span=20, adjust=False).mean().to_numpy()
    dif = s.ewm(span=12, adjust=False).mean() - s.ewm(span=26, adjust=False).mean()
    dea = dif.ewm(span=9, adjust=False).mean()
    hist = (2 * (dif - dea)).to_numpy()
    prev = np.r_[np.nan, c[:-1]]
    trend = (ma20 > ma60) & (ma60 > np.r_[np.full(20, np.nan), ma60[:-20]]) & (c > ma60)
    bias = c / ema20 - 1
    touched = (l <= ema20 * 1.02) & (c >= ema20 * 0.99)
    touch = pd.Series(touched).shift(1).rolling(10).max().to_numpy() == 1
    idx = pd.Series(np.where(touched, np.arange(len(c)), np.nan))
    last_touch = idx.shift(1).ffill().to_numpy()  # the most recent touch before each day
    last_touch = np.where(np.arange(len(c)) - last_touch <= 10, last_touch, np.nan)
    pb_low = pd.Series(l).shift(1).rolling(10).min().to_numpy()
    no_break = pd.Series(c < ema20 * 0.97).shift(1).rolling(10).max().to_numpy() == 0
    run = pd.Series(h).shift(1).rolling(20).max().to_numpy() >= ema20 * 1.08
    close5 = s.shift(1).rolling(5).max().to_numpy()
    s4 = trend & touch & no_break & run & (c / prev - 1 >= 0.02) & (c > close5) & (c > ma5) & (bias <= 0.12)
    s4 = s4 & ~(pd.Series(s4).shift(1).rolling(5).max().to_numpy() == 1)
    # a relaunch with no new EMA20 touch after the previous S4 day (within 10 sessions) is progress of that event,
    # not a new first entry; a fresh touch after it starts a new event
    progress, last_s4 = np.zeros(len(c), bool), None
    for t in np.nonzero(s4)[0]:
        progress[t] = last_s4 is not None and t - last_s4 <= 10 and not touched[last_s4 + 1:t].any()
        last_s4 = t
    n = len(c)
    high20 = s.rolling(20).max().to_numpy()
    diverge = np.zeros(n, bool)
    for t in np.nonzero(c >= high20)[0]:
        if t >= 45:
            k = t - 40 + int(np.argmax(c[t - 40:t - 4]))
            diverge[t] = c[t] > c[k] and hist[k] > 0 and hist[t] < hist[k]
    green_down = (hist < 0) & ~(hist > np.r_[np.nan, np.nan, hist[:-2]])
    dd60 = c / pd.Series(h).rolling(60).max().to_numpy() - 1
    vr = f["volume"].astype(float) / f["volume"].astype(float).shift(1).rolling(20).mean()
    return pd.DataFrame({"s4": s4, "progress": progress, "last_touch": last_touch, "pb_low": pb_low, "R1": diverge, "R2": green_down, "R3": dd60 < -0.15, "dd60": dd60, "bias": bias,
                         "ema20": ema20, "pct": c / prev - 1, "vr": vr.to_numpy(),
                         "macd": np.where(hist >= 0, "红柱", "绿柱")})

def clv(bar):
    """Close location of a daily bar: 0 at the low, 1 at the high."""
    rng = float(bar["high"]) - float(bar["low"])
    return round((float(bar["close"]) - float(bar["low"])) / rng, 3) if rng > 0 else None


def fundamentals(day, symbols, path=None):
    """Latest quarterly sales growth announced before day (the report of the latest announcement), per symbol."""
    path = Path(path or FIN_PATH)
    if not symbols or not path.exists():
        return {}, ""
    fin = pd.read_csv(path, usecols=["ts_code", "ann_date", "end_date", "q_sales_yoy"]).dropna(subset=["ann_date"])
    fin["ann"] = fin["ann_date"].astype(float).astype(int).astype(str)
    fin = fin[fin["ann"] < day.replace("-", "")]
    latest = fin["ann"].max() if len(fin) else ""
    fin["symbol"] = fin["ts_code"].str[-2:].str.lower() + "_" + fin["ts_code"].str[:6]
    last = fin[fin["symbol"].isin(symbols)].sort_values(["ann", "end_date"]).groupby("symbol").tail(1)
    return {s: (None if v != v else round(float(v), 2)) for s, v in zip(last["symbol"], last["q_sales_yoy"])}, latest


def reference(kept, sales):
    """Reference score among the day's recommendations: percentile ranks of high sales growth and low close location."""
    if not kept:
        return
    t = pd.DataFrame({"clv": [r.get("clv") for r in kept], "sales": [sales.get(r["symbol"]) for r in kept]}, dtype=float)
    score = ((-t["clv"]).rank(pct=True).fillna(0.5) + t["sales"].rank(pct=True).fillna(0.5)) / 2
    for r, s in zip(kept, score):
        r["q_sales_yoy"], r["ref_score"] = sales.get(r["symbol"]), round(float(s), 3)
    for i, r in enumerate(sorted(kept, key=lambda r: (-r["ref_score"], r["rank"])), 1):
        r["ref_rank"] = i

def pick(day, symbol, out_root=None, db=None):
    path = Path(out_root or SA.OUT_ROOT) / day.replace("-", "") / "s4_opportunities.json"
    if not path.exists():
        raise ValueError(f"{day} 还没有 S4 扫描结果，先运行扫描")
    out = json.loads(path.read_text(encoding="utf-8"))
    symbol = SA.normalize_symbol(symbol)
    if not any(r["symbol"] == symbol for r in out["recommended"]):
        reason = next(("、".join(r["excluded"]) for r in out["excluded"] if r["symbol"] == symbol), "不在当日 S4 推荐中")
        raise ValueError(f"{symbol} 不能按 S4 记入：{reason}")
    row = next(r for r in out["recommended"] if r["symbol"] == symbol)
    market = out.get("market_risk") or {}
    trend = out.get("index_trend") or {}
    if (out.get("as_of") != day or out.get("buy_advised") is not True
            or market.get("as_of") != day or market.get("level") != "NORMAL"):
        raise ValueError("S4 入场复核未通过：大盘红灯、未知或数据过期，不接受新加入")
    if trend.get("gate") != "ALLOW":
        raise ValueError("S4 入场复核未通过：中期趋势门未放行（" + str(trend.get("text"))[:120] + "）")
    mapping_path = ROOT / "concept_mapping.json"
    info = json.loads(mapping_path.read_text(encoding="utf-8")).get("mapping", {}).get(symbol, {}) if mapping_path.exists() else {}
    corrected, _ = TS.apply_membership_evidence({symbol: info}, day)
    info = corrected[symbol]
    risk = TR.evaluate_many([row.get("industry", "")] + info.get("concepts", []), day)
    if risk["theme_risk_action"] != "ALLOW":
        raise ValueError("S4 入场复核未通过：" + risk["theme_risk_reason"])
    ev = row.get("event") or {}
    return VP.add(symbol, day, db or VP.DB_PATH, strategy="S4_LAD",
                  meta={"s4_event_id": ev.get("event_id"), "s4_touch_day": ev.get("touch_day"),
                        "s4_pullback_low": ev.get("pullback_low"), "s4_signal_open": ev.get("signal_open")} if ev else None)


def main():
    parser = argparse.ArgumentParser()

    args = parser.parse_args()

POLICY_PATH = Path(__file__).with_name("calibration") / "route_policy.json"
STATUS_LABELS = {"CONFIRMATION_REVIEW": "突破接受，复核介入位置", "PREPARATION_WATCH": "准备观察，非买点",
                 "WAIT_PRICE": "趋势保留，偏离过大不追", "RISK_WATCH": "供给压力，暂缓介入",

                 "DATA_WAIT": "数据或资格不足"}
STATUS_ORDER = {s: i for i, s in enumerate(STATUS_LABELS)}


_BENCH = None


def _bench():
    """CSI300 20-day return by date, loaded once per process for the preparation detectors."""
    global _BENCH
    if _BENCH is None:
        try:
            _BENCH = PD.benchmark_ret20()
        except (OSError, ValueError, KeyError):
            _BENCH = pd.Series(dtype=float)
    return _BENCH


def load_policy() -> dict:
    return json.loads(POLICY_PATH.read_text(encoding="utf-8"))


def assess_routes(row: dict, policy: str, routes: list[str]) -> dict:
    result = RE.assess(row, policy)
    current_preparation = row.get("preparation_valid", row.get("prepared", False))
    independent = RC.price_confirmation(row)
    ready = independent or not (set(routes) & {"R2", "R3"}) or bool(current_preparation)
    result["independent_price_evidence"] = independent
    result["route_requirements_met"] = ready
    result["confirmation_pass"] = result["confirmation_pass"] and result["status"] == "CONFIRMATION_REVIEW"
    if not ready:
        result["confirmation_pass"] = False
        if result["status"] == "CONFIRMATION_REVIEW":
            result["status"] = "WAIT_ACCEPTANCE"
    return result


def snapshot(args: tuple[str, str, str]) -> dict:
    filename, date, code_hash = args
    path = Path(filename)
    stat = path.stat()
    fingerprint = [stat.st_size, stat.st_mtime_ns, code_hash, date]
    cache = Path("results/mainwave_engine/_route_cache") / date.replace("-", "") / f"{path.stem}.json"
    if cache.exists():
        cached = json.loads(cache.read_text(encoding="utf-8"))
        if cached.get("fingerprint") == fingerprint:
            return {**cached["record"], "cache_hit": True}
    symbol = path.stem.removesuffix("_daily")
    try:
        raw = pd.read_csv(path, encoding="utf-8-sig")
        raw = raw[pd.to_datetime(raw.date).le(date)]
        frame = RE.build(raw, symbol)
        if frame.empty:
            return {"symbol": symbol, "data_status": "SHORT_HISTORY", "cache_hit": False}
        row = frame.iloc[-1]
        names = ["date", "close", "ret5", "ret20", "bias_atr", "atr_pct", "probe_at", "retained", "supply",
                 "prepared", "prepared_at", "attempt_prepared_at", "prepared_before_attempt", "demand",
                 "supply_warning", "eligible", "st_known", "liquid", "amount_contract", "turnover_contract",
                 "turnover_relative", "range_contract", "breakout_status", "breakout_event",
                 "breakout_id", "breakout_attempt_at", "breakout_accepted_at", *RE.POLICIES]
        names.extend(RE.PT.FIELDS)
        record = json.loads(frame.tail(1)[list(dict.fromkeys(names))].to_json(orient="records", date_format="iso"))[0]
        record["date"] = str(row.date.date())
        record["symbol"] = symbol
        ema20 = frame.return_index.ewm(span=20, adjust=False).mean().iloc[-1]
        record["ema20_bias"] = float((row.return_index / ema20 - 1) * 100)
        record["data_status"] = "FRESH" if record["date"] == date else "STALE"
        accepted = frame.index[frame.breakout_id.eq(row.breakout_id) & frame.breakout_event.eq("ACCEPTED")]
        record["acceptance_age"] = int(len(frame) - 1 - accepted[-1]) if len(accepted) else None
        attempts = frame.index[frame.breakout_id.eq(row.breakout_id) & frame.breakout_event.eq("ATTEMPT")]
        record["attempt_age"] = int(len(frame) - 1 - attempts[-1]) if len(attempts) else None
        factor = float(row.close / row.return_index)
        record["boundary_price"] = float(row.breakout_boundary_index * factor) if pd.notna(row.breakout_boundary_index) else None
        record["invalid_price"] = float(row.breakout_invalid_index * factor) if pd.notna(row.breakout_invalid_index) else None
        record["frozen_atr_price"] = float(row.breakout_frozen_atr * factor) if pd.notna(row.breakout_frozen_atr) else None
        # Shadow preparation readings ride along in the cached record; they create no route and no candidate state.
        record["preparation"] = PD.latest(frame, _bench())
        record["cache_hit"] = False
        _write(cache, {"fingerprint": fingerprint, "record": record})
        return record
    except (OSError, ValueError, KeyError, pd.errors.ParserError) as exc:
        return {"symbol": symbol, "data_status": "ERROR", "error": str(exc), "cache_hit": False}


def route_ids(row: dict, chains: list[dict], date: str) -> tuple[list[str], list[dict]]:
    if row.get("data_status") != "FRESH" or not row.get("eligible", False):
        return [], []
    accepted = row.get("acceptance_age") is not None and row["acceptance_age"] <= 3 and row.get("breakout_status") in {"ACCEPTED", "ADVANCING"}
    attempt = row.get("attempt_age") is not None and row["attempt_age"] <= 3 and row.get("breakout_status") == "ATTEMPT"
    prepared = bool(row.get("prepared") or (accepted and row.get("prepared_before_attempt")))
    linked = [c for c in chains if str(c.get("as_of")) == date and c.get("theme_id") and c.get("leader")
              and c["leader"] != row["symbol"] and c.get("anchor_date") and str(c["anchor_date"]) <= date
              and c.get("theme_lifecycle") in {"FERMENTING", "IGNITED", "DIFFUSING"}
              and 1 <= pd.to_numeric(c.get("anchor_age_sessions"), errors="coerce") <= 8
              and pd.to_numeric(c.get("lag_gap_atr"), errors="coerce") >= 0.75]
    routes = []
    if prepared:
        routes.append("R2")
    if accepted or attempt:
        routes.append("R1")
    if linked:
        routes.insert(0, "R3")
        if "R1" in routes and not prepared:
            routes.remove("R1")
            routes.insert(0, "R1")
    return routes, linked


def run(date: str, workers: int = 4) -> dict:
    output = Path("results/mainwave_engine") / date.replace("-", "")
    policy = load_policy()
    code_hash = hashlib.sha256(b"".join(Path(p).read_bytes() for p in
        (__file__, RE.__file__, RE.PE.__file__, RE.BE.__file__, RE.PT.__file__, RC.__file__, PD.__file__))).hexdigest()
    paths = sorted(p for p in Path("data").glob("*_daily.csv") if re.fullmatch(
        r"(?:sh_(?:60|68)|sz_(?:00|30))\d{4}_daily.csv", p.name))
    started = perf_counter()
    audit = []
    with ProcessPoolExecutor(max_workers=workers) as pool:
        for i, row in enumerate(pool.map(snapshot, [(str(p), date, code_hash) for p in paths], chunksize=8), 1):
            audit.append(row)
            if i % 1000 == 0 or i == len(paths):
                print(f"[route_scan] {i}/{len(paths)} seconds={perf_counter()-started:.1f}", flush=True)
    chain_path = output / "slow_carrier_diagnostics.csv"
    chains = pd.read_csv(chain_path).fillna("").to_dict("records") if chain_path.exists() else []
    by_symbol = {}
    for chain in chains:
        by_symbol.setdefault(chain["symbol"], []).append(chain)
    mapping, mapping_meta = TS.load_mapping_as_of(date)
    records = []
    for row in audit:
        routes, linked = route_ids(row, by_symbol.get(row["symbol"], []), date)
        row["route_ids"] = "|".join(routes)
        if not routes:
            row["exclusion"] = row["data_status"] if row["data_status"] != "FRESH" else "NO_RECENT_ROUTE_EVIDENCE_OR_INELIGIBLE"
            continue
        result = assess_routes(row, policy["policy"], routes)
        identity = mapping.get(row["symbol"], {})
        theme = "|".join(dict.fromkeys(c["theme_label"] for c in linked)) or identity.get("industry", "未确认")
        path = {"R1": "IGNITION", "R2": "LURK", "R3": "PROPAGATION"}[routes[0]]
        records.append({**row, "as_of": date, "name": identity.get("name", row["symbol"]),
            "route_id": routes[0], "route_label": RE.ROUTE_LABELS[routes[0]], "route_status": result["status"],
            "route_status_label": STATUS_LABELS[result["status"]], "route_policy": policy["policy"],
            "route_requirements_met": result["route_requirements_met"],
            "independent_price_evidence": result["independent_price_evidence"],
            "strict_chain_verified": False,
            "chain_source_grade": "DECLARED_ASOF_ONLY" if linked else "NOT_APPLICABLE",
            "confirmation_pass": result["confirmation_pass"],
            "route_risks": "|".join(result["risks"]), "route_source": "DATED_CHAIN_AND_PRICE" if linked else "PRICE_RECONSTRUCTION_TODAY",
            "route_theme_id": "|".join(c["theme_id"] for c in linked),
            "discovery_path": path, "discovery_path_label": RE.ROUTE_LABELS[routes[0]],
            "belonging_theme": theme, "related_themes": theme, "theme_source": "DATED_CHAIN" if linked else "IDENTITY_ONLY_NOT_CATALYST",
            "theme_lifecycle": linked[0]["theme_lifecycle"] if linked else "",
            "stock_lifecycle": "COILED" if path == "LURK" else "IGNITION",
            "stock_lifecycle_label": "等待生命周期复核", "ema20_bias": row["ema20_bias"],
            "r5": row["ret5"] * 100, "r20": row["ret20"] * 100,
            "action": result["status"], "note": "因果量价代理；未宣称历史生产命中；无自动交易",
            "chain_validation": "DATED_ARCHIVE_SMALL_SAMPLE" if linked else "NOT_APPLICABLE"})
    records.sort(key=lambda r: (STATUS_ORDER[r["route_status"]], abs(r["bias_atr"]), r["symbol"]))
    for i, row in enumerate(records, 1):
        row["rank"] = i
    output.mkdir(parents=True, exist_ok=True)
    columns = ["symbol", "as_of", "route_id", "route_ids", "route_status", "rank"]
    frame = pd.DataFrame(records) if records else pd.DataFrame(columns=columns)
    frame.to_csv(output / "route_candidates.csv", index=False, encoding="utf-8-sig")
    publish(output / "route_candidates.csv", date)
    pd.DataFrame(audit).to_csv(output / "route_scan_audit.csv", index=False, encoding="utf-8-sig")
    publish(output / "route_scan_audit.csv", date)
    summary = {"as_of": date, "version": RE.VERSION, "policy": policy,
               "universe": len(paths), "fresh": sum(r["data_status"] == "FRESH" for r in audit),
               "errors": [r for r in audit if r["data_status"] == "ERROR"],
               "cache_hits": sum(r["cache_hit"] for r in audit), "candidates": len(records),
               "routes": {r: sum(r in row["route_ids"].split("|") for row in records) for r in RE.ROUTE_LABELS},
               "overlap_stocks": sum(len(r["route_ids"].split("|")) > 1 for r in records),
               "extra_route_memberships": sum(len(r["route_ids"].split("|")) - 1 for r in records),
               "data_status_counts": pd.Series([r["data_status"] for r in audit]).value_counts().to_dict(),
               "freshness_denominator": "CURRENT_LOCAL_UNIVERSE_NOT_EXPECTED_TRADABLE_UNIVERSE",
               "preparation_origins": frame.preparation_origin.fillna("UNKNOWN").replace("", "UNKNOWN").value_counts().to_dict() if len(frame) else {},
               "statuses": frame.route_status.value_counts().to_dict(), "mapping": mapping_meta,

    watch_columns = ["symbol", "name", "industry", "detector", "bias_atr", "excess20", "ret20", "above_ma20_now", "reclaim_failed", "breakout_status", "route_ids", "as_of", "basis"]
    pd.DataFrame(watch, columns=watch_columns).to_csv(output / "preparation_watch.csv", index=False, encoding="utf-8-sig")
    publish(output / "preparation_watch.csv", date)
    summary["preparation_watch"] = {"detector": PD.CHOSEN, "count": len(watch), "version": PD.VERSION}
    _write(output / "route_scan_summary.json", summary)
    fields = ["rank", "symbol", "name", "route_ids", "belonging_theme", "route_status_label", "preparation_first_seen", "preparation_origin", "probe_at",

    lines = [f"# 三路径证据扫描 {date}", "", f"- 全域 {len(paths)}，数据同日 {summary['fresh']}，候选 {len(records)}。",
             f"- {summary['routes']}；允许多路线命中，不重复计算股票数量。",
             "- R1=明确个股点火反查；R2=个股先行准备；R3=紧链传播承接。",
             "- 量价准备不是主力身份；偏离过大不是派发；确认复核不等于买入指令。",
             "- 同日数据率以本地股票域为分母，不是应交易股票覆盖率；缺日不自动解释为停牌。",

    (output / "route_scan.md").write_text("\n".join(lines), encoding="utf-8")
