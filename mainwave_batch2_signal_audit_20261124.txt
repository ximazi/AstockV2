"""Shared causal price evidence for daily discovery and historical experiments.

These are observable proxies, not claims about institutional ownership. Theme
propagation requires separate dated chain evidence; price alone cannot prove it.
"""
from __future__ import annotations

import numpy as np
import pandas as pd

from . import breakout_events as BE
from . import phase_events as PE
from . import preparation_track as PT

VERSION = "route_evidence_v2_20260910"
ROUTE_LABELS = {"R1": "明确个股点火反查", "R2": "个股先行准备", "R3": "紧链传播承接"}
# Fixed hypotheses before the broad historical run, not stock-specific fitted thresholds.
POLICIES = ("attempt", "accepted", "accepted_demand", "accepted_prepared",
            "accepted_bias2", "accepted_bias3", "accepted_bias4", "prepared_watch")


def build(raw: pd.DataFrame, symbol: str) -> pd.DataFrame:
    raw = raw.copy()
    raw["date"] = pd.to_datetime(raw.date)
    raw = raw.sort_values("date").drop_duplicates("date", keep="last")
    suspended = raw.loc[pd.to_numeric(raw.get("tradestatus", pd.Series(index=raw.index, dtype=float)), errors="coerce").eq(0)
                        & pd.to_numeric(raw.amount, errors="coerce").eq(0), "date"].dt.strftime("%Y-%m-%d").tolist()
    raw = raw[pd.to_numeric(raw.amount, errors="coerce").gt(0)].reset_index(drop=True)
    frame = PE.classify_frame(raw)
    if frame.empty:
        return frame
    identity = pd.DataFrame({"symbol": symbol, "date": frame.date.dt.strftime("%Y-%m-%d")})
    events = BE.annotate(frame, identity)
    frame = pd.concat([frame, events.drop(columns=["symbol", "date"])], axis=1)
    close, atr = frame.return_index, frame.atr20
    scale = close / frame.close
    for key in ("open", "high", "low"):
        frame[f"index_{key}"] = frame[key] * scale
    frame["index_close"] = close
    ma20 = close.rolling(20).mean()
    frame["bias_atr"] = (close - ma20) / atr
    frame["atr_pct"] = atr / close * 100
    frame["liquid"] = frame.amount.shift(1).rolling(20).median().ge(3e7)
    st = pd.to_numeric(frame.get("isST", pd.Series(np.nan, index=frame.index)), errors="coerce")
    frame["st_known"] = st.notna()
    frame["eligible"] = frame.liquid & st.eq(0) & atr.gt(0) & (frame.index >= 120)

    probe = frame.effort_result.eq("DEMAND_EXPANSION")
    # Preparation must predate the current session. All anchors expire, not infinite ffill.
    probe_base = close.shift(1).where(probe).shift(1).ffill(limit=20)
    probe_close = close.where(probe).shift(1).ffill(limit=20)
    probe_atr = atr.where(probe).shift(1).ffill(limit=20)
    frame["probe_at"] = frame.date.where(probe).shift(1).ffill(limit=20).dt.strftime("%Y-%m-%d").fillna("")
    frame["retained"] = (close.ge(probe_base + 0.5 * (probe_close - probe_base))
                         & close.ge(probe_close - 2 * probe_atr))
    turn = pd.to_numeric(frame.get("turnover", pd.Series(np.nan, index=frame.index)), errors="coerce").where(lambda s: s > 0)
    frame["turnover_relative"] = turn / turn.shift(1).rolling(20, min_periods=15).median()
    frame["turnover_contract"] = turn.rolling(5).median() / turn.shift(5).rolling(20).median()
    frame["amount_contract"] = frame.amount.rolling(5).median() / frame.amount.shift(5).rolling(20).median()
    width = frame.index_high - frame.index_low
    frame["range_contract"] = width.rolling(5).median() / width.shift(5).rolling(20).median()
    frame["supply"] = (frame.turnover_contract.le(0.85) & frame.amount_contract.le(0.85)
                       & frame.range_contract.le(0.85) & close.ge(frame.structure_floor)
                       & ma20.ge(ma20.shift(5)))
    frame["prepared"] = frame.retained & frame.supply & frame.bias_atr.between(-1, 2)
    prior_prepared = frame.prepared.shift(1, fill_value=False).rolling(30).max().eq(1)
    frame["prepared_at"] = frame.date.where(frame.prepared).shift(1).ffill(limit=30).dt.strftime("%Y-%m-%d").fillna("")
    at_attempt = frame.breakout_event.eq("ATTEMPT")
    frame["prepared_before_attempt"] = prior_prepared.astype(float).where(at_attempt).ffill().eq(1)
    frame["attempt_prepared_at"] = frame.prepared_at.astype("string").where(at_attempt).ffill().fillna("")
    frame["demand"] = (probe.rolling(3).sum().ge(1) & close.gt(close.shift(3)))
    stall = (frame.amount_ratio.ge(1.3) & frame.close_location.le(0.4)
             & (close.diff() <= 0.2 * atr))
    frame["supply_warning"] = frame.UTAD | frame.SOW | stall.rolling(5).sum().ge(2)
    frame["attempt"] = at_attempt
    frame["accepted"] = frame.breakout_event.eq("ACCEPTED")
    frame["accepted_demand"] = frame.accepted & frame.demand & ~frame.supply_warning
    frame["accepted_prepared"] = frame.accepted_demand & frame.prepared_before_attempt
    for n in (2, 3, 4):
        frame[f"accepted_bias{n}"] = frame.accepted_demand & frame.bias_atr.le(n)
    frame["prepared_watch"] = frame.prepared & ~frame.prepared.shift(1, fill_value=False)
    result = pd.concat([frame, PT.annotate(frame, symbol)], axis=1)
    result.attrs["suspended_days"] = suspended
    return result


def assess(row: dict, policy: str = "accepted_demand") -> dict:
    """Assess the current observation; policy is frozen separately after research."""
    if policy not in POLICIES or policy in {"attempt", "prepared_watch"}:
        raise ValueError(f"Not a confirmation policy: {policy}")
    accepted = row.get("breakout_status") in {"ACCEPTED", "ADVANCING"}
    risks = []
    if row.get("supply_warning", False):
        risks.append("SUPPLY_PRESSURE")
    if row.get("bias_atr", 0) > 3:
        risks.append("EXTENDED_NOT_DISTRIBUTION")
    passed = bool(row.get(policy, False))
    status = ("DATA_WAIT" if not row.get("eligible", False) else
              "RISK_WATCH" if "SUPPLY_PRESSURE" in risks else
              "WAIT_PRICE" if "EXTENDED_NOT_DISTRIBUTION" in risks else
              "CONFIRMATION_REVIEW" if passed else
              "PREPARATION_WATCH" if row.get("prepared", False) else
              "TREND_WATCH" if accepted else "WAIT_ACCEPTANCE")
    return {"version": VERSION, "policy": policy, "status": status,
            "risks": risks, "confirmation_pass": passed,
            "preparation_source": "CAUSAL_PRICE_PROXY", "orders_enabled": False}

"""Causal price-only breakout evidence; never changes production cycle states."""
from __future__ import annotations

import numpy as np
import pandas as pd

VERSION = "breakout_events_v2_20260910"
# Research placeholders in active sessions / frozen return-index ATR units.
LOOKBACK = 20
ACCEPT_CLOSES = 2
ACCEPT_TTL = 5
BREAK_BUFFER_ATR = 0.1
FAIL_BUFFER_ATR = 0.5
ADVANCE_ATR = 1.0
LABELS = {
    "NONE": "尚无突破事件", "ATTEMPT": "突破尝试，等待站稳",
    "ACCEPTED": "突破已接受，等待推进", "ADVANCING": "突破后已有价格推进",
    "REJECTED": "突破被拒绝", "EXPIRED": "突破尝试过期", "FAILED": "突破结构失效",
}


def annotate(frame: pd.DataFrame, timeline: pd.DataFrame) -> pd.DataFrame:
    scale = frame.return_index.to_numpy(float) / frame.close.to_numpy(float)
    high = frame.high.to_numpy(float) * scale
    low = frame.low.to_numpy(float) * scale
    close = frame.return_index.to_numpy(float)
    atr = frame.atr20.to_numpy(float)
    ceiling = pd.Series(high).rolling(LOOKBACK, min_periods=LOOKBACK).max().shift(1).to_numpy()
    floor = pd.Series(low).rolling(LOOKBACK, min_periods=LOOKBACK).min().shift(1).to_numpy()
    records, active = [], None
    for i, row in enumerate(timeline.to_dict("records")):
        date, event = row["date"], ""
        terminal = {"REJECTED", "EXPIRED", "FAILED"}
        if active and active["status"] not in terminal:
            trend_broken = (active["status"] in {"ACCEPTED", "ADVANCING"}
                            and i > 0 and close[i] < floor[i] - BREAK_BUFFER_ATR * atr[i - 1])
            if close[i] < active["invalid"] or trend_broken:
                active["status"] = "REJECTED" if active["status"] == "ATTEMPT" else "FAILED"
                event = active["status"]
            elif active["status"] == "ATTEMPT":
                active["holds"] = active["holds"] + 1 if close[i] > active["boundary"] else 0
                if i - active["start_index"] > ACCEPT_TTL:
                    active["status"], event = "EXPIRED", "EXPIRED"
                elif active["holds"] >= ACCEPT_CLOSES:
                    active.update(status="ACCEPTED", accepted_at=date, accepted_close=float(close[i]))
                    event = "ACCEPTED"
            elif active["status"] == "ACCEPTED" and close[i] >= active["accepted_close"] + ADVANCE_ATR * active["atr"]:
                active.update(status="ADVANCING", advanced_at=date)
                event = "ADVANCING"
        can_start = active is None or active["status"] in terminal
        if (can_start and not event and i >= LOOKBACK and np.isfinite(atr[i - 1]) and atr[i - 1] > 0
                and close[i] > ceiling[i] + BREAK_BUFFER_ATR * atr[i - 1]
                and close[i - 1] <= ceiling[i]):
            active = {"id": f"{row.get('symbol', '')}:breakout:{date}",
                      "status": "ATTEMPT", "attempt_at": date, "accepted_at": "", "advanced_at": "",
                      "boundary": float(ceiling[i]), "invalid": float(ceiling[i] - FAIL_BUFFER_ATR * atr[i - 1]),
                      "atr": float(atr[i - 1]), "start_index": i, "holds": 1}
            event = "ATTEMPT"
        a = active or {}
        status = a.get("status", "NONE")
        records.append({
            "breakout_id": a.get("id", ""), "breakout_status": status,
            "breakout_status_label": LABELS[status], "breakout_event": event,
            "breakout_attempt_at": a.get("attempt_at", ""), "breakout_accepted_at": a.get("accepted_at", ""),
            "breakout_advanced_at": a.get("advanced_at", ""), "breakout_boundary_index": a.get("boundary", np.nan),
            "breakout_invalid_index": a.get("invalid", np.nan), "breakout_frozen_atr": a.get("atr", np.nan),
            "breakout_version": VERSION,
        })
    return pd.concat([timeline.reset_index(drop=True), pd.DataFrame(records)], axis=1)

def recovery_evidence(a, j):
    close, low = a["index_close"][j], a["index_low"][j]
    healthy = a["eligible"][j] and not a["supply_warning"][j]
    recovery = (healthy and low >= a["index_low"][j - 1] and close > a["index_close"][j - 1]
                and a["close_location"][j] >= .5)
    strong = (recovery and close > a["index_high"][j - 1] and close >= a["ma20"][j]
              and a["close_location"][j] >= .6 and a["amount_ratio"][j] >= 1.)
    amount = float(np.median(a["amount"][max(0, j - 5):j]))
    contracted = amount > 0 and a["amount"][j] <= .85 * amount
    return bool(recovery), bool(strong), bool(contracted)


def order_ceiling(close, ma20, atr, initial_price=None):
    ceiling = min(close + .5 * atr, ma20 + 2 * atr)
    return min(ceiling, initial_price + .5 * atr) if initial_price is not None else ceiling


def reference(frame, calendar, symbol, setup=None):
    """Shared hypothetical entry anchor; not proof of a historical buy permission."""
    from . import data as D
    result = {"status": "NO_RECENT_ACCEPTANCE", "confirmations": [], "reference_basis": "HYPOTHETICAL_NEXT_OPEN_NOT_POSITION"}
    if frame.empty:
        return result
    frame = frame.reset_index(drop=True)
    days = frame.date.dt.strftime("%Y-%m-%d").tolist()
    accepted = frame.index[frame.accepted_bias2]
    i = int(accepted[-1]) if len(accepted) else -1
    setup = setup or {}
    confirmed_at = setup.get("setup_confirmed_at")
    from_setup = (setup.get("setup_status") in {"CONFIRMED", "HISTORY_CONFIRMED"} and confirmed_at in days
                  and days.index(confirmed_at) >= i and setup.get("setup_breakout_id")
                  == frame.breakout_id.iloc[days.index(confirmed_at)])
    if from_setup:
        i = days.index(confirmed_at)
    if i < 0:
        return result
    last = len(frame) - 1
    result.update(signal_date=days[i], breakout_id=str(frame.breakout_id.iloc[i]),
                  signal_kind=setup.get("setup_kind") if from_setup else "BREAKOUT_ACCEPTANCE")
    if i == last:
        return {**result, "status": "WAIT_INITIAL_FILL"}
    entry = i + 1
    clock = {day: n for n, day in enumerate(calendar)}
    if days[entry] not in clock or days[i] not in clock or days[last] not in clock:
        return {**result, "status": "DATA_GAP"}
    if clock[days[entry]] != clock[days[i]] + 1:
        return {**result, "status": "DATA_GAP"}
    age = clock[days[last]] - clock[days[entry]]
    result.update(reference_entry_date=days[entry], age_sessions=age)
    a = {c: frame[c].to_numpy() for c in frame.columns}
    a["ma20"] = frame.index_close.rolling(20).mean().to_numpy()
    price, atr, structure = a["index_open"][entry], a["atr20"][i], a["breakout_invalid_index"][i]
    scale = float(a["close"][last] / a["index_close"][last])
    if from_setup:
        atr = float(setup.get("setup_frozen_atr_price") or 0) / scale
        structure = float(setup.get("setup_invalid_price") or 0) / scale
    if not all(math.isfinite(x) and x > 0 for x in (price, atr, structure)):
        return {**result, "status": "NO_STRUCTURE"}
    ceiling = order_ceiling(a["index_close"][i], a["ma20"][i], atr)
    upper_limit = math.floor(a["preclose"][entry] * (1 + D.limit_pct(symbol) / 100) * 100 + .5) / 100
    if price > ceiling or price <= structure or a["open"][entry] >= upper_limit - .001 or a["isST"][entry] != 0:
        return {**result, "status": "INITIAL_REFERENCE_REJECTED"}
    result.update(reference_entry_price=float(price * scale), status="REFERENCE_READY",
                  _entry=entry, _signal=i, _price=float(price), _atr=float(atr), _structure=float(structure), _scale=scale)
    return result


def observe(frame, calendar, symbol, setup=None, *, anchor=None, exit_observation=None):
    """Conditional add evidence for the latest accepted episode, not a trade replay."""
    anchor = reference(frame, calendar, symbol, setup) if anchor is None else anchor
    result = {k: v for k, v in anchor.items() if not k.startswith("_")}
    if result.get("age_sessions", 0) >= WINDOW:
        return {**result, "status": "WINDOW_EXPIRED"}
    if result["status"] != "REFERENCE_READY":
        return result
    from . import exit_policy as XP
    if exit_observation is None:
        exit_observation = XP.observe(frame, calendar, symbol, anchor, XP.load_market(str(frame.date.iloc[-1].date())))
    if exit_observation["status"] == "DATA_GAP":
        return {**result, "status": "DATA_GAP"}
    if exit_observation.get("cancel_adds"):
        return {**result, "status": "SELL_REFERENCE_CANCELLED"}
    frame = frame.reset_index(drop=True)
    a = {c: frame[c].to_numpy() for c in frame.columns}
    a["ma20"] = frame.index_close.rolling(20).mean().to_numpy()
    days = frame.date.dt.strftime("%Y-%m-%d").tolist()
    clock = {day: n for n, day in enumerate(calendar)}
    i, entry, price, atr, structure, scale = (anchor[k] for k in ("_signal", "_entry", "_price", "_atr", "_structure", "_scale"))
    last = len(frame) - 1
    result["status"] = "WAIT_PULLBACK_RECLAIM"
    tests, stop = [None, None], price - 2 * atr
    suspended = set(frame.attrs.get("suspended_days", []))
    for j in range(entry, last + 1):
        missing = calendar[clock[days[j - 1]] + 1:clock[days[j]]] if days[j - 1] in clock else ["UNKNOWN"]
        if any(day not in suspended for day in missing):
            return {**result, "status": "DATA_GAP"}
        if (a["breakout_id"][j] != a["breakout_id"][i] or a["breakout_status"][j] not in {"ACCEPTED", "ADVANCING"}
                or a["index_close"][j] < structure):
            return {**result, "status": "STRUCTURE_INVALID"}
        if j > entry:
            _, strong, contracted = recovery_evidence(a, j)
            for k, pullback in enumerate(PULLBACKS):
                test = tests[k]
                if j == last and test and test["contracted"] and strong:
                    upper = order_ceiling(a["index_close"][j], a["ma20"][j], atr, price)
                    floor = max(structure, stop, exit_observation.get("soft_floor", stop * scale) / scale)
                    result["confirmations"].append({"tranche": k + 2, "budget_fraction": ADD_WEIGHTS[k],
                        "touch_date": test["day"], "confirm_date": days[j], "min_price": float(floor * scale),
                        "max_price": float(upper * scale), "execution": "NEXT_EXCHANGE_OPEN_ONLY"})
                if a["index_low"][j] <= price * (1 - pullback) and price * (1 - pullback) > stop:
                    if test is None:
                        tests[k] = {"day": days[j], "contracted": contracted}
                    elif contracted:
                        test["contracted"] = True
    if result["confirmations"]:
        result["status"] = ("CONDITIONAL_ADD_REVIEW" if any(c["max_price"] > c["min_price"] for c in result["confirmations"])
                            else "EXECUTION_BAND_EMPTY")
    return result

def market_frame(raw, as_of):
    f = raw.copy()
    f["date"] = pd.to_datetime(f.trade_date.astype(str))
    f = f[f.date.le(as_of)].sort_values("date").drop_duplicates("date")
    c = f.close
    ma20, ma60 = c.rolling(20).mean(), c.rolling(60).mean()
    f["market_strong"] = (c > ma20) & (ma20 > ma60) & (ma20 > ma20.shift(5))
    f["market_weak"] = (c < ma20) & (ma20 < ma20.shift(5))
    f["market_ret20"] = c.pct_change(20)
    return f.set_index("date")[["market_strong", "market_weak", "market_ret20"]]

def enrich(frame, market):
    f = frame.copy()
    c = f.index_close
    f["ma20"] = c.rolling(20).mean()
    f["low10"] = f.index_low.rolling(10).min()
    f["low20"] = f.index_low.rolling(20).min()
    f["ma60_exit"] = c.rolling(60).mean()
    f["ma20_up"] = f.ma20.gt(f.ma20.shift(5))
    joined = market.reindex(f.date)
    f["market_known"] = joined.market_ret20.notna().to_numpy()
    f["exit_strong"] = (f.market_known & joined.market_strong.eq(True).to_numpy()
                        & c.gt(f.ma20) & f.ma20.gt(f.ma60_exit) & f.ma20_up
                        & c.pct_change(20).ge(joined.market_ret20.to_numpy()))
    f["exit_market_weak"] = joined.market_weak.eq(True).to_numpy()
    width = (f.index_high - f.index_low).replace(0, np.nan)
    wick = (f.index_high - f[["index_close", "index_open"]].max(axis=1)) / width
    f["upper_wicks5"] = (wick.ge(.35) & f.close_location.le(.5)).rolling(5).sum()
    return f


def warning(a, j, entry, price, atr, prior_peak):
    """Close evidence schedules next-session action, not proof of fund intent."""
    cl, op, hi = (a[k][j] for k in ("index_close", "index_open", "index_high"))
    amount, location = a["amount_ratio"][j], a["close_location"][j]
    prior_low = a["low10"][j - 1]
    structural = cl < prior_low and cl < a["ma20"][j]
    heavy_break = structural and amount >= 1.3 and location <= .35
    failed_retest = (j >= entry + 2 and a["index_close"][j - 1] < a["low10"][j - 2]
                     and cl < a["low10"][j - 2] and cl < a["index_close"][j - 1]
                     and cl < a["ma20"][j])
    gap_break = structural and hi < a["index_low"][j - 1] and amount >= 1.3
    if heavy_break or failed_retest or gap_break:
        return "EXIT_STRUCTURE", {"heavy_break": bool(heavy_break), "failed_retest": bool(failed_retest),
                                  "gap_break": bool(gap_break)}
    mature = j - entry >= 5 and prior_peak >= price + 3 * atr
    near_peak = hi >= prior_peak - atr
    wick = PP.profit_warning(a, j, entry, price, atr, prior_peak)
    stall = near_peak and abs(cl - op) <= .25 * atr and amount >= 1.8 and location <= .5
    repeated = near_peak and a["upper_wicks5"][j] >= 3 and amount >= 1.3 and cl < a["index_close"][j - 1]
    if mature and (wick or stall or repeated):
        return "REDUCE_DISTRIBUTION_PROXY", {"weak_peak_close": bool(wick), "volume_stall": bool(stall),
                                             "repeated_wicks": bool(repeated)}
    return "", {}


def close_review(a, j, entry, price, atr, prior_peak, *, soft_stop, breaches=0, required=1,
                 pending_exit="", pending_reduce=False, reduced=False):
    """Shared EXTEND_CLOSE transition, evaluated strictly after the day's close."""
    cl = a["index_close"][j]
    if max(prior_peak, a["index_high"][j]) >= price + 2 * atr:
        if cl < soft_stop:
            if not breaches:
                required = 2 if any(a["exit_strong"][max(entry, j - 2):j]) else 1
            breaches += 1
            if breaches >= required:
                pending_exit = pending_exit or "SOFT_CLOSE_EXIT"
        else:
            breaches = 0
        trail = "low20" if a["exit_strong"][j] else "low10"
        soft_stop = max(soft_stop, float(a[trail][j]))
    action, evidence = warning(a, j, entry, price, atr, prior_peak)
    if action == "EXIT_STRUCTURE":
        pending_exit = pending_exit or action
        pending_reduce = False
    elif action and not reduced and not pending_exit:
        pending_reduce = True
    extension = None
    if j - entry + 1 == 20:
        extension = bool(a["exit_strong"][j] and cl >= price + 2 * atr and not pending_exit and not pending_reduce)
        if not extension:
            pending_exit = pending_exit or "TIME_REVIEW_EXIT"
    return {"soft_stop": float(soft_stop), "breaches": breaches, "required": required,
            "pending_exit": pending_exit, "pending_reduce": bool(pending_reduce),
            "action": action, "evidence": evidence, "extension": extension}


def observe(frame, calendar, symbol, anchor, market):
    """Audit one explicitly named hypothetical entry, without inferring actual holdings."""
    result = {"status": anchor["status"], "reference": {k: v for k, v in anchor.items() if not k.startswith("_") and k != "confirmations"},
              "events": [], "cancel_adds": False, "market_known": False, "strong": False}
    if frame.empty:
        return result
    f = enrich(frame.reset_index(drop=True), market)
    a = {c: f[c].to_numpy() for c in f.columns}
    days = f.date.dt.strftime("%Y-%m-%d").tolist()
    result.update(market_known=bool(f.market_known.iloc[-1]), strong=bool(f.exit_strong.iloc[-1]))
    # Position-free structural risk remains visible even without a usable entry.
    action, evidence = warning(a, len(f) - 1, 0, float("inf"), 1., 0.) if len(f) >= 21 else ("", {})
    result.update(current_structure_alert=action, current_structure_evidence=evidence)
    if anchor["status"] != "REFERENCE_READY":
        return result
    entry, price, atr, scale = (anchor[k] for k in ("_entry", "_price", "_atr", "_scale"))
    hard, soft, peak = price - 2 * atr, price - 2 * atr, price
    pending, reduction, reduced, hard_pending = "", False, False, False
    remaining, breaches, required = 1., 0, 1
    targets = [(price + 3 * atr, .5), (price + 6 * atr, .25)]
    clock, suspended = {day: n for n, day in enumerate(calendar)}, set(frame.attrs.get("suspended_days", []))
    result.update(status="HOLD_REFERENCE", hard_floor=float(hard * scale), soft_floor=float(soft * scale))
    for j in range(entry, len(f)):
        if days[j] not in clock or days[j - 1] not in clock or any(
                day not in suspended for day in calendar[clock[days[j - 1]] + 1:clock[days[j]]]):
            return {**result, "status": "DATA_GAP", "cancel_adds": True, "last_evaluated_date": days[j - 1]}
        op, hi, lo = (a[k][j] for k in ("index_open", "index_high", "index_low"))
        if not all(math.isfinite(a[k][j]) and a[k][j] > 0 for k in
                   ("index_open", "index_high", "index_low", "index_close", "preclose")):
            return {**result, "status": "DATA_GAP", "cancel_adds": True, "last_evaluated_date": days[j - 1]}
        lower = math.floor(a["preclose"][j] * (1 - D.limit_pct(symbol) / 100) * 100 + .5) / 100
        raw_high = a["high"][j] if "high" in a else hi * a["close"][j] / a["index_close"][j]
        locked = raw_high <= lower + .001
        events = result["events"]
        if j > entry and not locked and reduction and not hard_pending and not pending and op > hard:
            events.append({"date": days[j], "reason": "RISK_REDUCTION", "reference_fraction": remaining / 2})
            remaining /= 2
            reduction, reduced = False, True
        reason = ""
        if j == entry:
            hard_pending = lo <= hard
        elif not locked and (hard_pending or pending or lo <= hard):
            reason = "DEFERRED_STOP" if hard_pending else pending or "INITIAL_STOP"
        elif lo <= hard:
            hard_pending = True
        if not reason and j > entry and not locked:
            for level, weight in list(targets):
                if hi >= level:
                    used = min(weight, remaining)
                    events.append({"date": days[j], "reason": "PARTIAL_TARGET", "reference_fraction": used})
                    remaining -= used
                    targets.remove((level, weight))
            if remaining <= 1e-10:
                reason = "TARGET_COMPLETED"
        if not reason and j - entry + 1 >= 60:
            if not locked:
                reason = "HORIZON"
            else:
                pending = pending or "HORIZON"
        result.update(held_sessions=j - entry + 1, last_evaluated_date=days[j], locked=bool(locked))
        if reason:
            if remaining > 1e-10:
                events.append({"date": days[j], "reason": reason, "reference_fraction": remaining})
            return {**result, "status": "REFERENCE_CLOSED", "closed_at": days[j], "reason": reason, "cancel_adds": True}
        review = close_review(a, j, entry, price, atr, peak, soft_stop=soft, breaches=breaches,
                              required=required, pending_exit=pending, pending_reduce=reduction, reduced=reduced)
        soft, breaches, required = (review[k] for k in ("soft_stop", "breaches", "required"))
        pending, reduction = review["pending_exit"], review["pending_reduce"]
        peak = max(peak, hi)
        if review["extension"] is not None:
            result["extension_review"] = {"date": days[j], "extend": review["extension"]}
        result.update(soft_floor=float(soft * scale), soft_breaches=breaches, soft_required=required,
                      pending_exit=pending, pending_reduce=reduction, evidence=review["evidence"],
                      status=("HARD_EXIT_PENDING" if hard_pending else "EXIT_NEXT_OPEN" if pending else
                              "REDUCE_NEXT_OPEN" if reduction else "SOFT_CONFIRMING" if breaches else "HOLD_REFERENCE"),
                      cancel_adds=bool(events or hard_pending or pending or reduction))
    return result

"""Recall audit of the discovery layer against labelled TREND episodes.

Target: results/mainwave_episodes_2026.csv (ex-post label: +25% within 40 sessions without a -15% path,
liquid; class TREND = neither lottery ladder nor an extended run). Detectors: the replayable route
evidence (R2 preparation, R1 attempt/acceptance) and the stock lifecycle stages, rebuilt from bars only —
both are prefix-stable, so a reading at date d equals what the engine could have known at d.

For every episode start t the audit asks: was the stock eligible; which detector fired first inside
[t-10, t+5] sessions; how extended the price already was when it fired. For every detector firing in the
period it also asks whether an episode followed within 10 sessions (precision). Misses and late hits are
the recall gaps ranking weights cannot fix."""
from __future__ import annotations

import argparse
import json
import statistics
from collections import Counter, defaultdict
from datetime import datetime
from pathlib import Path

import pandas as pd

from mainwave_engine import route_evidence as RE, stock_lifecycle as SL

ROOT = Path(".")
DATA = ROOT / "data"
OUT = ROOT / "results/research/screening_recall_audit"
BEFORE, AFTER = 10, 5
PRECISION_WINDOW = 10
LEAD_BUCKETS = (("EARLY", -99, -1), ("ON_TIME", 0, 2), ("LATE", 3, 5))
DETECTORS = ("R2_prepared", "R1_attempt", "R1_accepted", "R1_accepted_demand", "LC_ready", "LC_ignition")


def bucket(lead):
    for name, lo, hi in LEAD_BUCKETS:
        if lo <= lead <= hi:
            return name
    return "MISS"


def detectors(frame, timeline):
    """Per-session boolean firings for each detector, aligned on the route frame's dates."""
    dates = frame.date.dt.strftime("%Y-%m-%d")
    stage = {}
    if timeline is not None and not timeline.empty:
        stage = dict(zip(pd.to_datetime(timeline.date).dt.strftime("%Y-%m-%d"), timeline.cycle_stage))
    fired = {"R2_prepared": frame.prepared.fillna(False).astype(bool).tolist(),
             "R1_attempt": frame.breakout_event.eq("ATTEMPT").tolist(),
             "R1_accepted": frame.breakout_event.eq("ACCEPTED").tolist(),
             "R1_accepted_demand": frame.accepted_demand.fillna(False).astype(bool).tolist(),
             "LC_ready": [stage.get(d) in ("ARMED", "READY") for d in dates],
             "LC_ignition": [stage.get(d) == "IGNITION" for d in dates]}
    return dates.tolist(), fired


def audit_symbol(symbol, starts, span):
    path = DATA / f"{symbol}_daily.csv"
    if not path.exists():
        return [{"symbol": symbol, "start": s, "status": "NO_BARS"} for s in starts], []
    raw = pd.read_csv(path, encoding="utf-8-sig")
    frame = RE.build(raw, symbol)
    if frame.empty:
        return [{"symbol": symbol, "start": s, "status": "SHORT_HISTORY"} for s in starts], []
    try:
        _, timeline = SL.analyze(symbol, str(frame.date.iloc[-1].date()), DATA)
    except Exception:  # lifecycle is optional evidence; its absence is recorded, not fatal
        timeline = None
    dates, fired = detectors(frame, timeline)
    index = {d: i for i, d in enumerate(dates)}
    eligible = frame.eligible.fillna(False).astype(bool).tolist()
    bias = frame.bias_atr.tolist()
    rows = []
    for start in starts:
        i = index.get(start)
        if i is None:
            rows.append({"symbol": symbol, "start": start, "status": "START_NOT_IN_BARS"})
            continue
        lo, hi = max(0, i - BEFORE), min(len(dates) - 1, i + AFTER)
        row = {"symbol": symbol, "start": start, "status": "ELIGIBLE" if eligible[i] else "INELIGIBLE_AT_START",
               "stage_eve": (timeline.cycle_stage.iloc[index[dates[i - 1]]] if timeline is not None and not timeline.empty and i >= 1 and dates[i - 1] in index and index[dates[i - 1]] < len(timeline) else None)}
        first_any = None
        for det in DETECTORS:
            hit = next((j for j in range(lo, hi + 1) if fired[det][j]), None)
            lead = (hit - i) if hit is not None else None
            row[f"{det}_lead"] = lead
            row[f"{det}_bucket"] = bucket(lead) if lead is not None else "MISS"
            row[f"{det}_bias"] = round(float(bias[hit]), 2) if hit is not None and pd.notna(bias[hit]) else None
            if hit is not None and (first_any is None or hit < first_any[1]):
                first_any = (det, hit)
        row["first_detector"] = first_any[0] if first_any else None
        row["first_lead"] = (first_any[1] - i) if first_any else None
        row["first_bucket"] = bucket(first_any[1] - i) if first_any else "MISS"
        row["first_bias"] = round(float(bias[first_any[1]]), 2) if first_any and pd.notna(bias[first_any[1]]) else None
        rows.append(row)
    # precision: every firing inside the audit span → did an episode start within the next 10 sessions?
    starts_idx = sorted(index[s] for s in starts if s in index)
    firings = []
    lo_span, hi_span = span
    for det in ("R2_prepared", "R1_accepted", "R1_accepted_demand", "LC_ready", "LC_ignition"):
        prev = False
        for j, on in enumerate(fired[det]):
            new = on and not prev
            prev = on
            if not new or not (lo_span <= dates[j] <= hi_span) or not eligible[j]:
                continue
            followed = any(j <= s <= j + PRECISION_WINDOW for s in starts_idx)
            firings.append({"symbol": symbol, "detector": det, "date": dates[j], "episode_within_10": followed,
                            "bias_atr": round(float(bias[j]), 2) if pd.notna(bias[j]) else None})
    return rows, firings


def run(episodes_path="results/mainwave_episodes_2026.csv", cls="TREND", limit=None, min_gain=None, out_root=OUT):
    episodes = pd.read_csv(episodes_path)
    episodes = episodes[episodes.cls == cls]
    if min_gain is not None:
        episodes = episodes[episodes.max_gain >= min_gain]
    if limit:
        episodes = episodes.sample(n=min(limit, len(episodes)), random_state=20260923)
    span = (str(episodes.start.min()), str(episodes.start.max()))
    by_symbol = defaultdict(list)
    for r in episodes.itertuples():
        by_symbol[r.stem].append(str(r.start))
    rows, firings = [], []
    started = datetime.now()
    for n, (symbol, starts) in enumerate(sorted(by_symbol.items()), 1):
        r, f = audit_symbol(symbol, sorted(starts), span)
        rows.extend(r)
        firings.extend(f)
        if n % 200 == 0 or n == len(by_symbol):
            print(f"[recall] {n}/{len(by_symbol)} symbols, {len(rows)} episodes, {(datetime.now() - started).seconds}s", flush=True)
    frame = pd.DataFrame(rows)
    meta = episodes[["stem", "start", "name", "board", "industry", "max_gain", "min_path", "ret60_prior", "bias60"]].rename(columns={"stem": "symbol"})
    meta["start"] = meta.start.astype(str)
    frame = frame.merge(meta, on=["symbol", "start"], how="left")
    fires = pd.DataFrame(firings)
    out = Path(out_root) / datetime.now().strftime("%Y%m%d_%H%M%S")
    out.mkdir(parents=True, exist_ok=True)
    frame.to_csv(out / "episodes.csv", index=False, encoding="utf-8-sig")
    fires.to_csv(out / "firings.csv", index=False, encoding="utf-8-sig")
    summary = summarize(frame, fires, span, cls)
    (out / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=1, default=str), encoding="utf-8")
    (out / "report.md").write_text(report(summary), encoding="utf-8")
    print(f"[recall] written {out}")
    return out, summary


def summarize(frame, fires, span, cls):
    audited = frame[frame.status.isin(["ELIGIBLE", "INELIGIBLE_AT_START"])]
    eligible = audited[audited.status == "ELIGIBLE"]
    def buckets(col):
        return dict(Counter(eligible[col].fillna("MISS")))
    per_detector = {det: {"buckets": buckets(f"{det}_bucket"),
                          "hit_rate": round(float(eligible[f"{det}_bucket"].ne("MISS").mean()), 3) if len(eligible) else None,
                          "median_lead": (float(eligible[f"{det}_lead"].median()) if eligible[f"{det}_lead"].notna().any() else None),
                          "median_bias_at_hit": (float(eligible[f"{det}_bias"].median()) if eligible[f"{det}_bias"].notna().any() else None)}
                    for det in DETECTORS}
    any_hit = eligible.first_bucket.ne("MISS")
    extended_at_hit = eligible.first_bias.gt(2.0)
    month = eligible.start.str[:7]
    by_month = {m: {"episodes": int(len(g)), "any_hit": round(float(g.first_bucket.ne("MISS").mean()), 3),
                    "early_or_on_time": round(float(g.first_bucket.isin(["EARLY", "ON_TIME"]).mean()), 3)} for m, g in eligible.groupby(month)}
    by_board = {b: {"episodes": int(len(g)), "any_hit": round(float(g.first_bucket.ne("MISS").mean()), 3)} for b, g in eligible.groupby("board")}
    precision = {}
    if len(fires):
        for det, g in fires.groupby("detector"):
            precision[det] = {"firings": int(len(g)), "episode_within_10": round(float(g.episode_within_10.mean()), 3),
                              "median_bias": float(g.bias_atr.median()) if g.bias_atr.notna().any() else None}
    return {"class": cls, "span": span, "episodes_total": int(len(frame)), "audited": int(len(audited)), "eligible": int(len(eligible)),
            "ineligible_at_start": int((audited.status == "INELIGIBLE_AT_START").sum()),
            "status_counts": dict(Counter(frame.status)), "any_hit_rate": round(float(any_hit.mean()), 3) if len(eligible) else None,
            "first_bucket": dict(Counter(eligible.first_bucket)), "first_detector": dict(Counter(eligible.first_detector.fillna("NONE"))),
            "extended_at_first_hit_share": round(float(extended_at_hit[any_hit].mean()), 3) if any_hit.any() else None,

def wave_failure_pairs(f,waves):
    """Same-band failed executions, for morphology only (chosen by observed outcome)."""
    failed=f[f.mature & f.status.eq("CLOSED") & f.net.le(0)].groupby(["date","board","band"])
    ids=set(zip(waves.symbol,waves.date))
    used=set()
    rows=[]
    for r in waves[waves.status.eq("CLOSED")].sort_values(["date","symbol"]).itertuples(index=False):
        key=r.date,r.board,r.band
        if key not in failed.groups:
            continue
        peers=failed.get_group(key)
        peers=peers[peers.amount20.between(.5*r.amount20,2*r.amount20)]
        options=[(abs(np.log(t.amount20/r.amount20)),t.symbol,t)
                 for t in peers.itertuples(index=False) if (t.symbol,t.date) not in ids|used]
        if not options:
            continue
        peer=min(options,key=lambda x:x[:2])[2]
        used.add((peer.symbol,peer.date))
        fields=("bias_atr","opening_bias","opening_gap_atr","risk_pct","acceptance_delay","pulse_delay","net")
        rows.append(dict(date=r.date,board=r.board,band=r.band,market=r.market,symbol=r.symbol,peer=peer.symbol,
                         **{prefix+k:getattr(t,k) for prefix,t in (("wave_",r),("peer_",peer)) for k in fields}))
    return pd.DataFrame(rows)


def pulse_open_diagnostic(waves,cal,limits):
    """Price paid for waiting on linked accepted waves, never an early-entry backtest."""
    pos={d:i for i,d in enumerate(cal)}
    rows=[]
    pulse=waves.pulse.fillna("")
    candidates=waves[waves.opened.eq(True)&pulse.ne("")&pulse.lt(waves.date)]
    for s,g in candidates.groupby("symbol"):
        a=read_raw(s).set_index("date")
        # Use the same cumulative return scale so corporate-action nominal gaps are not price improvement.
        r=a.pct_chg.div(100).fillna(a.close.pct_change()).fillna(0).clip(-.95,3.)
        ix=a.open/a.close*np.exp(np.log1p(r).cumsum())
        for w in g.itertuples(index=False):
            d=cal[pos[w.pulse]+1]
            state="MISSING_PULSE_NEXT_OPEN"
            delta=None
            if d in a.index and w.entry_open_day in a.index:
                row=a.loc[d]
                upper,_,_=band(s,SimpleNamespace(day=d,preclose=row.preclose,isST=row.isST),limits)
                state=("NO_TRADE_AT_PULSE_OPEN" if row.tradestatus!=1 or not row.amount>0 or not row.open>0
                       else "PULSE_LIMIT_UNVERIFIED" if not np.isfinite(upper)
                       else "PULSE_OPEN_LIMIT_UP" if row.open>=upper else "COMPARABLE_OPEN")
                if state=="COMPARABLE_OPEN" and ix[d]>0:
                    delta=float(100*(ix[w.entry_open_day]/ix[d]-1))
            rows.append(dict(symbol=s,start=w.start,pulse=w.pulse,accepted=w.date,pulse_open_date=d,
                             accepted_open_date=w.entry_open_day,status=state,open_cost_delta_pct=delta))
    return pd.DataFrame(rows)


def pair_intervals(pairs,cal):
    """20-session block bootstrap; do not treat several stocks on the same date as IID."""
    rows=[]
    rng=np.random.default_rng(20261119)
    for label,g in (("all_markets",pairs),("green_only",pairs[pairs.market.eq("GREEN")])):
        for year,x in g.groupby(g.date.str[:4]):
            days=[d for d in cal if x.date.min()<=d<=x.date.max()]
            block=min(20,len(days))
            y=x.groupby("date").net_delta.agg(["sum","size"]).reindex(days,fill_value=0)
            starts=rng.integers(len(days),size=(3000,int(np.ceil(len(days)/block))))
            take=((starts[:,:,None]+np.arange(block))%len(days)).reshape(3000,-1)[:,:len(days)]
            sims=y["sum"].to_numpy()[take].sum(axis=1)/np.maximum(1,y["size"].to_numpy()[take].sum(axis=1))
            low,high=np.quantile(sims,[.025,.975])
            rows.append(dict(sample=label,year=year,n=len(x),days=x.date.nunique(),mean=x.net_delta.mean(),
                             median=x.net_delta.median(),lower95=low,upper95=high,
                             near_better=x.net_delta.gt(0).mean()*100))
    return pd.DataFrame(rows)


def supplements(out, f, audit, pairs, cal, limits):
    wave_failure_pairs(f,audit).to_csv(out/"wave_failure_pairs.csv",index=False)
    pulse_open_diagnostic(audit,cal,limits).to_csv(out/"pulse_open_timing.csv",index=False)
    pair_intervals(pairs,cal).to_csv(out/"pair_intervals.csv",index=False)

"""Account replay picking S4 candidates by the features that passed research/s4_selection_features.py (fixed there
before running): signal-day close location (lower is better) and latest quarterly sales growth (higher is better).
Score of a candidate = mean of its within-day percentile ranks among the candidates entering that day (equal weights,
nothing fitted; a missing feature counts 0.5). Account 5 x 20% with the G1 entry filter, 100 seeds, as in
research/account_replay.py; compared with random picks, each feature alone, and the reverse composite (sanity check).
python research/s4_selection_replay.py -> results/research/s4_selection/replay.csv
"""

sys.path.insert(0, str(Path(__file__).resolve().parent))

def scores(trades, feat, cols):
    """Within the candidates sharing an entry day: percentile rank per feature (direction applied), averaged."""
    t = pd.DataFrame({"i": range(len(trades)), "symbol": [x["symbol"] for x in trades], "date": [x["signal"] for x in trades],
                      "entry": [x["entry"] for x in trades]})
    t = t.merge(feat, on=["symbol", "date"], how="left")
    parts = []
    for col, sign in cols:
        parts.append((sign * t[col]).groupby(t.entry).rank(pct=True).fillna(0.5))
    s = sum(parts) / len(parts)
    return dict(zip(t.i, s))


def main():
    feat = pd.read_csv(OUT / "features.csv", encoding="utf-8-sig")[["symbol", "date", "clv", "q_sales_yoy"]]
    trades = AR.s4_trades()
    bench = pd.read_csv(ROOT / "data/000852_SH_index.csv")
    bench["d"] = pd.to_datetime(bench.trade_date).dt.strftime("%Y-%m-%d")
    first = min(x["entry"] for x in trades)
    last = max(max(fd for fd, _, _ in x["fills"]) for x in trades)
    calendar = [d for d in bench.d if first <= d <= last]
    variants = {"随机挑选": None,
                "组合分（收盘位置低 + 营收增速高）": scores(trades, feat, [("clv", -1), ("q_sales_yoy", 1)]),
                "只按收盘位置低": scores(trades, feat, [("clv", -1)]),
                "只按营收增速高": scores(trades, feat, [("q_sales_yoy", 1)]),
                "反向组合（对照）": scores(trades, feat, [("clv", 1), ("q_sales_yoy", -1)])}
    rows = []
    for name, sc in variants.items():
        runs = []
        for seed in range(AR.SEEDS):
            nav, ex, done = AR.run_account(trades, 5, 0.20, True, seed, calendar, score=sc)
            runs.append(AR.metrics(nav, ex, done))
        m = pd.DataFrame(runs)
        rec = {"挑选方式": name}
        for col in ("总收益%", "2025%", "2026%", "最大回撤%", "成交笔数", "胜率%", "盈亏比", "每笔期望%", "最长连亏"):
            rec[col] = round(m[col].median(), 2)
        rec["总收益 5%~95%"] = f"{m['总收益%'].quantile(0.05):.1f} ~ {m['总收益%'].quantile(0.95):.1f}"
        rec["优于随机中位的种子%"] = None
        rows.append(rec)
        if name == "随机挑选":
            rand_median = m["总收益%"].median()
        rec["优于随机中位的种子%"] = round((m["总收益%"] > rand_median).mean() * 100)
    r = pd.DataFrame(rows)

    print(r.to_string(index=False))
